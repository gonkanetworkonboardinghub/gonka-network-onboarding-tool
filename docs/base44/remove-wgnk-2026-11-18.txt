async function gnkPrice() {
  const tryOne = async (name, fn) => {
    try {
      const usd = await fn();
      return usd > 0 && isFinite(usd) ? { name, usd } : null;
    } catch (_) { return null; }
  };
  const sources = (await Promise.all([
    tryOne("CoinGecko", async () =>
      Number((await getJson("https://api.coingecko.com/api/v3/simple/price?ids=gonka&vs_currencies=usd")).gonka.usd)),
    tryOne("CoinPaprika", async () =>
      Number((await getJson("https://api.coinpaprika.com/v1/tickers/gnk-gonka")).quotes.USD.price))
    // Not here on purpose: crypto.com and Bitget publish a GNK price page but
    // have no GNK market to quote, and SafeTrade — the one order book that does
    // list it — answers 403 to anything that is not a browser. Its price reaches
    // us through CoinGecko, which tracks that listing.
  ])).filter(Boolean);

  if (!sources.length) return { usd: null, sources: [] };
  const sorted = sources.map((s) => s.usd).sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  const usd = sorted.length % 2 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
  return { usd, sources };
}

let reading = null;   // { at, sources }
export function readMarkets() {
  const keep = reading && reading.sources && reading.sources.length ? 5 * 60 * 1000 : 60 * 1000;
  if (!reading || Date.now() - reading.at > keep) {
    const r = { at: Date.now(), sources: null };
    r.pending = gnkPrice().then((p) => (r.sources = p.sources || [])).catch(() => (r.sources = []));
    reading = r;
  }
  return reading.pending;
}

/**
 * When does the next Proof of Compute round start? The chain publishes the
 * exact start block (epoch_stages.next_poc_start); converting to wall-clock
 * time just needs the real block rate, measured from two block timestamps.
 * Epochs are ~23h, so the start time drifts earlier every day — this is the
 * only reliable way to plan a launch window.
 */
/** Current network tip height — cheap enough to poll from the Verify page. */
async function chainHeight(seed) {
  const now = await getJson(seed + "/chain-rpc/block");
  // time: when that block was made, for the countdown's quick look (pocEta).
  return { height: Number(now.result.block.header.height), time: Date.parse(now.result.block.header.time) || null };
}

/**
 * How long until block `nextStart`, from the newest block and the pace blocks
 * are being made at now.
 *
 * The next block is not a whole block away: part of it has already passed
 * since the newest one was made. That part is taken off, by this computer's
 * clock, and never more than one block's worth, so a clock that is minutes
 * wrong cannot throw the countdown off by more than a few seconds.
 */
function pocEta({ height, time, nextStart, secPerBlock, now = Date.now() }) {
  const blocksLeft = nextStart - height;
  if (!(blocksLeft > 0)) return 0;
  const sinceTip = time ? Math.min(secPerBlock * 1000, Math.max(0, now - time)) : 0;
  return Math.max(0, blocksLeft * secPerBlock * 1000 - sinceTip);
}

async function nextPoc(seed) {
  const k = K.get();
  const j = await getJson(seed + k.api.latestEpoch);
  const now = await getJson(seed + "/chain-rpc/block");
  const h = Number(now.result.block.header.height);
  const tNow = Date.parse(now.result.block.header.time);
  const back = Math.max(1, h - 5000);
  const old = await getJson(seed + "/chain-rpc/block?height=" + back);
  const tOld = Date.parse(old.result.block.header.time);
  const secPerBlock = (tNow - tOld) / 1000 / (h - back);
  const nextStart = Number(j.epoch_stages && j.epoch_stages.next_poc_start);
  if (!nextStart || !isFinite(secPerBlock) || secPerBlock <= 0) throw new Error("Epoch data unavailable");
  const startBlock = Number(j.epoch_stages && j.epoch_stages.poc_start) || 0;
  const blocksLeft = nextStart - h;
  // The pace over the last eight hours says how long an epoch is. How long
  // the blocks still to come will take is better told by the pace of late:
  // measured over about as many blocks as are left (300 to 5,000 of them).
  let secPerBlockNow = secPerBlock;
  const span = Math.min(5000, Math.max(300, blocksLeft));
  if (blocksLeft > 0 && span < 5000 && h - span > 1) {
    try {
      const near = await getJson(seed + "/chain-rpc/block?height=" + (h - span));
      const v = (tNow - Date.parse(near.result.block.header.time)) / 1000 / span;
      if (isFinite(v) && v > 0) secPerBlockNow = v;
    } catch (_) { /* the long measure will do */ }
  }
  return {
    phase: j.phase || "",
    epoch: Number((j.epoch_stages && j.epoch_stages.epoch_index) || (j.latest_epoch && j.latest_epoch.index)) || null,
    blocksLeft,
    etaMs: pocEta({ height: h, time: tNow, nextStart, secPerBlock: secPerBlockNow }),
    secPerBlock,
    // For the countdown's quick looks between two full readings (app.js).
    secPerBlockNow, nextStart, height: h,
    // for the epoch progress bar: how far along the current epoch is
    totalBlocks: startBlock ? nextStart - startBlock : 0
  };
}

/**
 * For every reference node-config-*.json file currently in the repo, reads
 * which GPU class it targets (from the filename) and how many GPUs each
 * model in it needs (from its --tensor-parallel-size vLLM arg) — read
 * directly from GitHub, so it reflects whatever the repo has right now
 * without needing a server connection or an app update.
 * Returns rows: { file, gpuClass, model, gpuCount }.
 */
/**
 * Do the Docker images this deployment needs actually have a build for the
 * server's CPU architecture? Gonka has shipped releases (0.2.14) published
 * arm64-only, which makes an x86 host fail deep into the launch sequence with
 * confusing errors — a failed pull, or a chain node that starts and dies.
 * Checking the registry first turns 20 wasted minutes into one clear message.
 * Returns [{ image, archs, ok }] for images that could be checked.
 */
async function checkImageArchs(images, wantArch = "amd64") {
  const out = [];
  for (const ref of images) {
    const m = /^ghcr\.io\/([^/]+)\/([^:@]+)(?::([^@]+))?/.exec(ref);
    if (!m) continue;                       // only ghcr images are checkable here
    const [, owner, name, tag] = m;
    if (!tag || ref.includes("@sha256:")) continue;   // digest-pinned: skip
    try {
      const tk = await fetch(`https://ghcr.io/token?scope=repository:${owner}/${name}:pull`,
        { signal: AbortSignal.timeout(8000) });
      const { token } = await tk.json();
      const res = await fetch(`https://ghcr.io/v2/${owner}/${name}/manifests/${tag}`, {
        headers: {
          Authorization: `Bearer ${token}`,
          Accept: "application/vnd.oci.image.index.v1+json,application/vnd.docker.distribution.manifest.list.v2+json"
        },
        signal: AbortSignal.timeout(10000)
      });
      if (!res.ok) continue;
      const idx = await res.json();
      let archs = (idx.manifests || [])
        .map((x) => x.platform && x.platform.architecture)
        .filter((a) => a && a !== "unknown");

      // A single-architecture manifest has no platform list — the arch lives in
      // the config blob. Gonka has switched between multi-arch indexes and
      // single-arch images, and skipping the latter meant an arm64-only image
      // would pass silently, which is exactly the case this check exists for.
      if (!archs.length && idx.config && idx.config.digest) {
        try {
          const cfgRes = await fetch(`https://ghcr.io/v2/${owner}/${name}/blobs/${idx.config.digest}`, {
            headers: { Authorization: `Bearer ${token}` },
            redirect: "follow",
            signal: AbortSignal.timeout(10000)
          });
          if (cfgRes.ok) {
            const cfg = await cfgRes.json();
            if (cfg.architecture) archs = [cfg.architecture];
          }
        } catch (_) { /* leave unknown rather than guessing */ }
      }
      if (!archs.length) continue;          // genuinely undeterminable — don't guess
      out.push({ image: ref, archs: [...new Set(archs)], ok: archs.includes(wantArch) });
    } catch (_) { /* registry unreachable — never block setup on this check */ }
  }
  return out;
}

/**
 * Can a server of this architecture run the CURRENT release at all? Reads the
 * compose files straight from GitHub (no server needed), then checks the
 * registry. This runs on the Welcome page so someone learns "x86 can't join
 * right now" BEFORE renting hardware — not twenty minutes into a paid launch.
 */
async function releaseArchStatus(wantArch = "amd64") {
  const k = K.get();
  const gh = parseGithubRepo(k.repo.url);
  if (!gh) return null;
  const inRepoPath = k.repo.joinDir.split("/").slice(1).join("/");
  const ref = k.repo.checkout || k.repo.branch;
  const base = `https://raw.githubusercontent.com/${gh.owner}/${gh.repo}/${ref}/${inRepoPath}`;
  const images = new Set();
  for (const f of ["docker-compose.yml", "docker-compose.mlnode.yml"]) {
    try {
      const res = await fetch(`${base}/${f}`, { signal: AbortSignal.timeout(10000) });
      if (!res.ok) continue;
      const text = await res.text();
      for (const m of text.matchAll(/^\s*image:\s*(\S+)/gm)) images.add(m[1]);
    } catch (_) { /* offline — caller treats null as "unknown", never blocks */ }
  }
  if (!images.size) return null;
  const results = await checkImageArchs([...images], wantArch);
  if (!results.length) return null;
  return { arch: wantArch, checked: results.length, bad: results.filter((r) => !r.ok) };
}

async function repoGpuConfigs() {
  const k = K.get();
  const gh = parseGithubRepo(k.repo.url);
  if (!gh) return [];
  // joinDir's first segment is the local clone directory name (== repo name),
  // an artifact of `git clone` with no destination arg — not part of the
  // repo's own tree, which is what GitHub's contents API expects.
  const inRepoPath = k.repo.joinDir.split("/").slice(1).join("/");
  const url = `https://api.github.com/repos/${gh.owner}/${gh.repo}/contents/${inRepoPath}?ref=${k.repo.branch}`;
  const res = await fetch(url, {
    headers: { "User-Agent": "gonka-setup-wizard" },
    signal: AbortSignal.timeout(10000)
  });
  if (!res.ok) throw new Error(`GitHub API error: ${res.status}`);
  const list = await res.json();
  // Includes the plain node-config.json (no GPU-class suffix) too, so its
  // model reference gets cross-checked against governance approval just
  // like the classed files do.
  const files = (list || []).filter((f) => /^node-config(-.*)?\.json$/.test(f.name));

  const rows = [];
  await Promise.all(files.map(async (f) => {
    if (!f.download_url) return;
    const cls = k.gpuClasses.find((c) => c.match.test(f.name));
    try {
      const cfgRes = await fetch(f.download_url, { signal: AbortSignal.timeout(10000) });
      const cfg = await cfgRes.json();
      for (const node of cfg) {
        for (const [model, def] of Object.entries(node.models || {})) {
          const args = def.args || [];
          const idx = args.indexOf("--tensor-parallel-size");
          const gpuCount = idx >= 0 ? parseInt(args[idx + 1], 10) : null;
          rows.push({ file: f.name, gpuClass: cls ? cls.id : null, model, gpuCount });
        }
      }
    } catch (_) { /* skip files that fail to fetch/parse */ }
  }));
  return rows;
}

function decToNumber(d) {
  // { value, exponent } decimal encoding used by inference params
  return Number(d.value) * Math.pow(10, Number(d.exponent));
}

async function myEpochWeight(seed, address) {
  if (!address) return null;
  const j = await getJson(seed + "/chain-api/productscience/inference/inference/current_epoch_group_data");
  const list = (j && j.epoch_group_data && j.epoch_group_data.validation_weights) || [];
  const mine = list.find((x) => x.member_address === address);
  if (!mine) return null;
  return {
    weight: Number(mine.weight) || 0,
    confirmationWeight: Number(mine.confirmation_weight) || 0,
    reputation: Number(mine.reputation) || 0
  };

async function recommendCollateral(seed, myWeight = null) {
  const k = K.get();
  const params = await getJson(seed + k.api.inferenceParams);
  const cp = params.params.collateral_params;
  const baseRatio = decToNumber(cp.base_weight_ratio);
  const perUnit = decToNumber(cp.collateral_per_weight_unit);

  let weight = myWeight;
  if (weight == null) {
    const ep = await epochParticipants(seed);
    const list = ep?.active_participants?.participants || [];
    weight = Math.max(...list.map((p) => Number(p.weight) || 0), 0);
  }
  const deposit = Math.round(weight * (1 - baseRatio) * perUnit * k.collateralBufferX);
  return {
    weightUsed: weight,
    baseWeightRatio: baseRatio,
    collateralPerUnit: perUnit,
    buffer: k.collateralBufferX,
    depositNgonka: deposit
  };
}

/**
 * Balance in ngonka for an address. A brand-new wallet has no on-chain
 * account until it receives funds — the chain answers 404, which means
 * "0 balance", not "couldn't check". null is reserved for real network
 * failures where the balance is genuinely unknown.
 */
async function balanceOf(seed, address) {
  const k = K.get();
  // Primary source: the chain's own bank module. The /v2/accounts index only
  // knows addresses that are already participants, so it 404s for a freshly
  // funded wallet — which made the funding watcher report 0 forever.
  try {
    const bank = await getJson(seed + "/chain-api/cosmos/bank/v1beta1/balances/" + address);
    const coins = (bank && bank.balances) || [];
    const n = coins.find((c) => /ngonka/i.test(c.denom));
    if (n) return Number(n.amount);
    if (Array.isArray(coins)) return 0;   // account exists, holds nothing
  } catch (_) { /* fall through to the participant index */ }

  let acc;
  try { acc = await getJson(seed + k.api.accounts + address); }
  catch (e) {
    if (/HTTP 404/.test(String(e && e.message))) return 0;
    return null;
  }
  const bal = acc.balances || acc.balance || acc.coins || null;
  if (Array.isArray(bal)) {
    const n = bal.find((b) => /ngonka/i.test(b.denom));
    return n ? Number(n.amount) : 0;
  }
  if (bal && bal.amount != null) return Number(bal.amount);
  if (typeof acc.balance === "number") return acc.balance;
  return 0;
}

/**
 * Probe a public URL from THIS machine — a genuine outside-in reachability
 * test for the user's node (any HTTP response, even an error page, proves
 * the port is open; only a connection failure/timeout means unreachable).
 */
async function probeUrl(url, timeout = 8000) {
  try {
    const res = await fetch(url, { signal: AbortSignal.timeout(timeout), redirect: "manual" });
    return { reachable: true, status: res.status };
  } catch (e) {
    return { reachable: false, error: String((e && e.cause) || e).slice(0, 200) };
  }
}

/**
 * What is left of the fee allowance a wallet gave its node's key, as the chain
 * has it now. The node pays its fees out of the wallet through this allowance,
 * so what it can spend is the wallet's balance capped by it. Pure: takes the
 * answer of cosmos/feegrant/v1beta1/issued/{wallet}.
 *   { found: false }                      the wallet gave no allowance
 *   { found: true, left, expires, over }  left in ngonka, null when unlimited;
 *                                         over when it has run out of time
 * With several, the one with the most left counts (as Monitor/Vote shows it).
 */
function readAllowance(answer, now = Date.now()) {
  const list = ((answer && answer.allowances) || []).map((a) => {
    // Basic, or a periodic / message-limited one wrapped around a basic one.
    let inner = (a && a.allowance) || {};
    if (inner.allowance) inner = inner.allowance;
    if (inner.basic) inner = inner.basic;
    const limit = Array.isArray(inner.spend_limit) ? inner.spend_limit : [];
    const coin = limit.find((c) => /ngonka/i.test(c.denom));
    const over = !!inner.expiration && !(new Date(inner.expiration).getTime() > now);
    return { found: true, left: limit.length ? Number((coin || {}).amount || 0) : null, expires: inner.expiration || null, over };
  });
  if (!list.length) return { found: false };
  const worth = (x) => (x.over ? -1 : x.left === null ? Infinity : x.left);
  return list.sort((a, b) => worth(b) - worth(a))[0];
}
async function allowanceLeft(seed, address) {
  return readAllowance(await getJson(seed + "/chain-api/cosmos/feegrant/v1beta1/issued/" + address));
}

/**
 * earnings.js — what a given rig actually mines on Gonka right now.
 *
 * Everything here is measured, not modelled:
 *   - how much GNK the chain minted over the last full epoch (supply now vs
 *     supply one epoch ago), which is what gets shared out;
 *   - the weight each paid participant holds this epoch
 *     (current_epoch_group_data.validation_weights — these sum to total_weight,
 *     the number rewards are split by);
 *   - which GPUs are behind that weight, because every participant publishes
 *     its hardware (hardware_nodes) and each ml node's Proof of Compute weight.
 *
 * Divide those and you get GNK per GPU per day for each GPU class, observed
 * from nodes actually running — no benchmark, no promise. The configurations
 * come from the same place as the wizard's first step (the Gonka repo's
 * node-config files), so the table people see here is the table they can
 * actually deploy.
 *
 * Nothing here is a forecast: earnings fall when more weight joins, a fresh
 * node ramps up over its first epochs, and the GNK price moves. The UI says so.
 */
const netdata = require("./netdata");
const K = require("../knowledge");

const GPU_CLASSES = [
  [/B300/i, "B300"], [/B200/i, "B200"], [/H200/i, "H200"], [/H100/i, "H100"], [/A100/i, "A100"],
  // Hosts run these and are paid for them, but Gonka publishes no setup for
  // them, so they are listed with a † only. (\bH20\b leaves H200 alone.)
  [/\bH20\b/i, "H20"], [/RTX\s*5090/i, "RTX 5090"], [/RTX\s*4090/i, "RTX 4090"],
  [/RTX\s*PRO\s*6000/i, "RTX PRO 6000"], [/MI355X/i, "MI355X"], [/MI300X/i, "MI300X"]
];
/**

 * copies of a setup as the server has room for (mlnode runner.py: GPUs ÷ GPUs
 * per copy), so an 8-GPU server with a 4-GPU setup runs two and earns twice.
 */
const FULL_SERVER = 8;
/** A class+model pair needs at least this many GPUs behind it to be quoted on its own. */
const MIN_SAMPLE_GPUS = 8;

const median = (a) => {
  const v = [...a].sort((x, y) => x - y);
  if (!v.length) return 0;
  return v.length % 2 ? v[(v.length - 1) / 2] : (v[v.length / 2 - 1] + v[v.length / 2]) / 2;
};

const classOf = (type) => (GPU_CLASSES.find(([re]) => re.test(String(type))) || [, null])[1];

async function getJson(url, timeout = 15000, headers = {}) {
  const res = await fetch(url, { headers, signal: AbortSignal.timeout(timeout) });

}

/**
 * The GNK price, from as many independent places as answer in time; the median
 * is used, so one stale or odd source cannot swing it. Only markets for native
 * GNK are read (both of these track GNK/USDT on SafeTrade). The Ethereum token
 * is left out on purpose since the exploit of 18 November 2026: its price no
 * longer stands for GNK.
 */
async function gnkPrice() {
  const tryOne = async (name, fn) => {
    try {
      const usd = await fn();
      return usd > 0 && isFinite(usd) ? { name, usd } : null;
    } catch (_) { return null; }
  };
  const sources = (await Promise.all([
    tryOne("CoinGecko", async () =>
      Number((await getJson("https://api.coingecko.com/api/v3/simple/price?ids=gonka&vs_currencies=usd")).gonka.usd)),
    tryOne("CoinPaprika", async () =>
      Number((await getJson("https://api.coinpaprika.com/v1/tickers/gnk-gonka")).quotes.USD.price))
    // Not here on purpose: crypto.com and Bitget publish a GNK price page but
    // have no GNK market to quote, and SafeTrade — the one order book that does
    // list it — answers 403 to anything that is not a browser. Its price reaches
    // us through CoinGecko, which tracks that listing.
  ])).filter(Boolean);

  if (!sources.length) return { usd: null, sources: [] };
  const sorted = sources.map((s) => s.usd).sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  const usd = sorted.length % 2 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
  return { usd, sources };
}

/** Minting, weight and the GPUs behind it, for the epoch that just paid out. */
async function chainFacts(seed) {
  const base = String(seed).replace(/\/+$/, "");
  // Gonka's public nodes refuse (HTTP 503) when asked many questions in a row:
  // in October 2026 node2 refused 13 of 28 hosts' hardware asked back to back,
  // and every refused host had silently dropped out of the figures. So a
  // refused or failed question is asked again of the next public node, after
  // a pause; only "not found" is an answer.
  const nodes = [base, ...K.get().seedNodes.map((s) => String(s).replace(/\/+$/, "")).filter((s) => s !== base)];
  const api = async (p, height) => {
    let last = null;
    for (let i = 0; i < nodes.length * 2; i++) {
      try { return await getJson(nodes[i % nodes.length] + p, 20000, height ? { "x-cosmos-block-height": String(height) } : {}); }
      catch (e) {
        last = e;
        if (/ -> 404$/.test(e.message)) throw e;
        await new Promise((r) => setTimeout(r, i + 1 < nodes.length ? 300 : 2000));
      }
    }
    throw last;
  };

  const head = Number((await api("/chain-rpc/status")).result.sync_info.latest_block_height);
  const params = (await api("/chain-api/productscience/inference/inference/params")).params;
  const epochLen = Number(params.epoch_params.epoch_length);

  // Real elapsed time, not the nominal block count: epochs drift.
  const tNow = new Date((await api("/chain-rpc/block")).result.block.header.time).getTime();
  const tPrev = new Date((await api("/chain-rpc/block?height=" + (head - epochLen))).result.block.header.time).getTime();
  const epochHours = (tNow - tPrev) / 3.6e6;

  const supplyNow = BigInt((await api("/chain-api/cosmos/bank/v1beta1/supply/by_denom?denom=ngonka")).amount.amount);
  const supplyPrev = BigInt((await api("/chain-api/cosmos/bank/v1beta1/supply/by_denom?denom=ngonka", head - epochLen)).amount.amount);
  const mintedGnk = Number(supplyNow - supplyPrev) / 1e9;

  // Rewards do not land in one piece: each epoch's award is cut into slices
  // released one per epoch, so what you mine today arrives over months.
  let vestingEpochs = Number(params.tokenomics_params && params.tokenomics_params.reward_vesting_period) || 0;
  try {
    const sv = await api("/chain-api/productscience/inference/streamvesting/params");
    vestingEpochs = Number(sv.params.reward_vesting_period) || vestingEpochs;
  } catch (_) { /* the tokenomics param above is the same number */ }

  const egd = (await api("/chain-api/productscience/inference/inference/current_epoch_group_data")).epoch_group_data;
  const totalWeight = Number(egd.total_weight);
  const memberWeight = new Map((egd.validation_weights || []).map((m) => [m.member_address, Number(m.weight)]));
  const active = (await api("/v1/epochs/current/participants")).active_participants;

  // Per node, not pooled: a few stragglers (a node that joined mid-epoch, or
  // spent part of it failing validation) drag a pooled average well below what
  // a healthy node of the same kind earns, so the median is the honest answer
  // to "what would mine get".
  const byClass = new Map();        // "H100"        -> { gpus, perGpu: [] }
  const byClassModel = new Map();   // "H100|model"  -> { gpus, perGpu: [] }
  // How many machines run exactly this setup now, and for how many hosts:
  // "H100|model|8" -> { machines, hosts }. Only machines with one kind of GPU.
  const sizes = new Map();
  const bump = (map, key, gpus, weight, nodeGpus, host) => {
    const cur = map.get(key) || { gpus: 0, perGpu: [], nodeGpus: [], hosts: new Set(), byHost: new Map() };
    cur.gpus += gpus;
    if (gpus > 0) {
      cur.perGpu.push(weight / gpus); cur.nodeGpus.push(nodeGpus || gpus); cur.hosts.add(host);
      (cur.byHost.get(host) || cur.byHost.set(host, []).get(host)).push(weight / gpus);
    }
    map.set(key, cur);
  };

  // Every machine with its slice of the weight, before anything is counted.
  const machines = [];
  for (const p of active.participants || []) {
    const paid = memberWeight.get(p.index);
    if (!paid) continue;                       // not in the paid set this epoch
    // A node can appear under several models; its PoC weight is the same node.
    const poc = new Map();
    for (const group of p.ml_nodes || []) for (const n of group.ml_nodes || []) {
      const w = Number(n.poc_weight || 0);
      if (w > 0) poc.set(n.node_id, Math.max(w, poc.get(n.node_id) || 0));
    }
    const pocTotal = [...poc.values()].reduce((a, b) => a + b, 0);
    if (!pocTotal) continue;

    let answer = null;
    try {
      answer = await api("/chain-api/productscience/inference/inference/hardware_nodes/" + p.index);
    } catch (e) {
      if (/ -> 404$/.test(e.message)) continue;   // no published hardware: it just isn't counted
      // Not one public node would say: figures without this host would be wrong, so none are given.
      throw new Error(`No public Gonka node would tell ${p.index}'s hardware just now (${e.message}).`);
    }
    const hardware = (answer && answer.nodes && answer.nodes.hardware_nodes) || [];
    const byId = new Map(hardware.map((n) => [n.local_id, n]));

    for (const [nodeId, w] of poc) {
      const node = byId.get(nodeId);
      if (!node || !(node.hardware || []).length) continue;
      const share = (w / pocTotal) * paid;     // this node's slice of what the member is paid for
      const gpus = node.hardware.reduce((a, h) => a + Number(h.count || 0), 0) || 1;
      // Its kind of GPU: the one it has most of.
      const kind = classOf([...node.hardware].sort((a, b) => Number(b.count || 0) - Number(a.count || 0))[0].type);
      machines.push({ host: p.index, node, share, gpus, kind, perGpu: share / gpus });
    }
  }

  // Hosts report their own hardware and the network does not check it. In
  // October 2026 three of the largest hosts reported each of their machines as
  // a single H100 while the machine carried the weight of dozens of GPUs
  // (17,000 to 79,000 Proof of Compute weight, where a real 4 × H100 has 3,400
  // to 5,300): taken at their word, they made a 4 × H100 look fifteen times
  // better than it is. A machine whose weight per GPU is more than
  // LEFT_OUT_ABOVE times the middle machine of its kind cannot be the hardware
  // it reports, so it is left out of the figures — and counted, so the screen
  // can say so.
  const typicalOfKind = new Map();
  for (const m of machines) if (m.kind) (typicalOfKind.get(m.kind) || typicalOfKind.set(m.kind, []).get(m.kind)).push(m.perGpu);
  for (const [kind, list] of typicalOfKind) typicalOfKind.set(kind, median(list));
  const leftOut = new Map();                   // kind -> { machines, hosts }
  for (const { host, node, share, gpus, kind, perGpu } of machines) {
    const typical = kind ? typicalOfKind.get(kind) : 0;
    if (typical > 0 && perGpu > typical * LEFT_OUT_ABOVE) {
      const cur = leftOut.get(kind) || { machines: 0, hosts: new Set() };
      cur.machines++; cur.hosts.add(host);
      leftOut.set(kind, cur);
      continue;
    }
    const models = (node.models || []).length ? node.models : [null];
    const only = node.hardware.length === 1 ? classOf(node.hardware[0].type) : null;
    if (only) for (const model of models) {
      if (!model) continue;
      const key = `${only}|${model}|${gpus}`;
      const cur = sizes.get(key) || { machines: 0, hosts: new Set() };
      cur.machines++; cur.hosts.add(host);
      sizes.set(key, cur);
    }
    for (const h of node.hardware) {
      const cls = classOf(h.type);
      if (!cls) continue;
      const count = Number(h.count) || 0;
      const part = share * (count / gpus);
      bump(byClass, cls, count, part, gpus, host);
      for (const model of models) {
        if (!model) continue;
        bump(byClassModel, cls + "|" + model, count / models.length, part / models.length, gpus, host);
      }
    }
  }

  return {
    epochIndex: Number(egd.epoch_index),
    epochHours,
    mintedGnk,
    totalWeight,
    // What the network is actually serving this epoch. A model nobody runs has
    // no observed weight, so quoting earnings for it would be a guess.
    runningModels: egd.sub_group_models || [],
    vestingEpochs,
    participants: (egd.validation_weights || []).length,
    gnkPerWeight: totalWeight ? mintedGnk / totalWeight : 0,
    rewardParams: params.bitcoin_reward_params || null,
    api,
    byClass, byClassModel, sizes, leftOut,
    // Hosts with any machine left out above: the same rule (3 x the middle machine of
    // its kind) by which the network page marks a host that reports far fewer GPUs
    // than its weight takes. Their other machines still count in the figures; they
    // are only kept out of the range.
    overReporters: new Set([...leftOut.values()].flatMap((x) => [...x.hosts]))
  };
}

/**
 * What a working host is paid, against its full share of an epoch's reward
 * (what was minted × its weight ÷ all the weight). A host that missed too much
 * work, or was left out, is paid nothing, and its share goes to the governance
 * account rather than to the others — so it says nothing about what a working
 * rig earns, yet it had been pulling every figure down by 10–40%, depending on
 * how many others failed that epoch. The hosts that were paid got about 90% of
 * their full share in every epoch measured (Oct 2026). The median over the hosts
 * paid in the last three finished epochs, from epochs.json and each epoch's
 * weights. Null when it cannot be read or does not add up.
 */
/**
 * Every host's pay for the recent epochs (epochs.json). The data job hands over
 * the file it has just read (GNOT_EPOCHS_FILE), so its figures are never one run
 * behind; the app reads the published copy. Null when there is neither.
 */
async function epochsRecord() {
  // (typeof: the website's copy of this file runs where there may be no process)
  const local = typeof process !== "undefined" && process.env ? process.env.GNOT_EPOCHS_FILE : "";
  if (local) return JSON.parse(require("fs").readFileSync(local, "utf8"));
  const url = K.get().epochsUrl;
  return url ? getJson(url, 20000) : null;
}

async function typicalShare(facts) {
  const bp = facts.rewardParams;
  if (!bp || !bp.use_bitcoin_rewards) return null;
  try {
    const file = await epochsRecord();
    if (!file) return null;
    const hosts = file.hosts || [];
    const done = (file.epochs || [])
      .filter((e) => e.e < facts.epochIndex && (e.r || []).some((r) => Number(r[1]) > 0))
      .sort((a, b) => b.e - a.e).slice(0, 3);
    const decay = Number(bp.decay_rate && bp.decay_rate.value) * Math.pow(10, Number(bp.decay_rate && bp.decay_rate.exponent));
    const ratios = [], epochs = [];
    for (const ep of done) {
      const g = await facts.api(`/chain-api/productscience/inference/inference/epoch_group_data/${ep.e}`);
      const vws = (g && g.epoch_group_data && g.epoch_group_data.validation_weights) || [];
      const all = vws.reduce((a, v) => a + Math.max(0, Number(v.weight) || 0), 0);
      const minted = Number(bp.initial_epoch_reward) / 1e9 * Math.exp(decay * (ep.e - Number(bp.genesis_epoch)));
      if (!(all > 0) || !(minted > 0)) continue;
      const got = new Map(ep.r.map((r) => [hosts[r[0]], Number(r[1] || 0) / 1e9]));
      for (const v of vws) {
        const full = Math.max(0, Number(v.weight) || 0) / all * minted, paid = got.get(v.member_address) || 0;
        if (full > 0 && paid > 0) ratios.push(paid / full);
      }
      epochs.push(ep.e);
    }
    if (ratios.length < 5) return null;
    ratios.sort((a, b) => a - b);
    const share = ratios.length % 2 ? ratios[(ratios.length - 1) / 2] : (ratios[ratios.length / 2 - 1] + ratios[ratios.length / 2]) / 2;
    // Outside this range means a misread file or a rule the code here does not know.
    if (!(share > 0.3 && share < 1.05)) return null;
    return { share, epochs, hosts: ratios.length };
  } catch (_) { return null; }
}

/**
 * What the hosts were actually paid in the last finished epoch, from the
 * record of every host's rewarded coins that the data job publishes
 * (epochs.json): between 59% and 89% of what was minted in October 2026. Shown
 * as it is, and used for the figures only when typicalShare() cannot be read —
 * most of the gap is hosts paid nothing, which a working rig is not. Null when
 * the record cannot be read or does not add up.
 */
async function lastPaid(currentEpoch, mintedGnk) {
  try {
    const file = await epochsRecord();
    if (!file) return null;
    const done = (file.epochs || [])
      .filter((e) => e.e < currentEpoch && (e.r || []).some((r) => Number(r[1]) > 0))
      .sort((a, b) => b.e - a.e)[0];
    if (!done) return null;
    const gnk = done.r.reduce((a, r) => a + Number(r[1] || 0), 0) / 1e9;
    const share = mintedGnk ? gnk / mintedGnk : 0;
    // A share outside this range means a half-collected epoch or a misread file.
    if (!(share > 0.2 && share < 1.05)) return null;
    return { epoch: done.e, gnk, share, hosts: done.r.filter((r) => Number(r[1]) > 0).length };
  } catch (_) { return null; }
}

/**
 * Per deployable configuration: the models and GPU counts the wizard's first
 * step offers, priced with what that class of GPU is earning today.
 */
async function estimate(seed, { force = false } = {}) {
  if (!force && cache && Date.now() - cache.at < CACHE_MS) return cache.data;

  const [facts, price, repoRows] = await Promise.all([
    chainFacts(seed),
    gnkPrice(),
    netdata.repoGpuConfigs().catch(() => [])
  ]);
  const repoKeys = new Set(repoRows.map((r) => r.model + "|" + r.gpuClass));
  const rows = repoRows.concat(
    (K.get().curatedConfigs || []).filter((c) => !repoKeys.has(c.model + "|" + c.gpuClass)));
  const perDay = facts.epochHours ? 24 / facts.epochHours : 0;
  // A rig's share of the network's weight now, times what is minted, times the
  // part of its full share a working host is actually paid. Without that part,
  // what all hosts were paid last epoch; without either, what was minted.
  const paid = await lastPaid(facts.epochIndex, facts.mintedGnk);
  const typical = await typicalShare(facts);
  if (typical && facts.totalWeight) facts.gnkPerWeight = facts.mintedGnk / facts.totalWeight * typical.share;
  else if (paid && facts.totalWeight) facts.gnkPerWeight = paid.gnk / facts.totalWeight;

  // Machines reporting the same setup earn very different amounts (Nov 2026: one
  // 8 x H200 at about 600 GNK a day, others of that kind three to four times as
  // much): hosts report their own hardware and the network does not check it, and
  // tuning and time on the network differ. So each figure, the middle machine,
  // also carries where half the hosts running it fall: each host by its middle
  // machine, the lowest and highest quarter of hosts left aside, and hosts that
  // report far fewer GPUs than their weight takes left out. Lowest-to-highest was
  // tried first and was 3-5 x wide, each end a single host (the user, 2026-11-10).
  const quantile = (sorted, q) => {
    const i = (sorted.length - 1) * q, lo = Math.floor(i), hi = Math.ceil(i);
    return sorted[lo] + (sorted[hi] - sorted[lo]) * (i - lo);
  };
  const spread = (entry) => {
    const perHost = [...entry.byHost.entries()].filter(([h]) => !facts.overReporters.has(h)).map(([, list]) => median(list)).sort((a, b) => a - b);
    const all = entry.byHost.size;
    return perHost.length ? { lo: quantile(perHost, 0.25), hi: quantile(perHost, 0.75), n: perHost.length, all } : { lo: 0, hi: 0, n: 0, all };
  };
  const rangeOf = (lo, hi, n, gpuCount) => (n >= 4 && hi > lo ? {
    low: lo * gpuCount * facts.gnkPerWeight * perDay,
    high: hi * gpuCount * facts.gnkPerWeight * perDay,
    hosts: n
  } : null);
  const hug = (r, mid) => (r ? { ...r, low: Math.min(r.low, mid), high: Math.max(r.high, mid) } : null);
  const weightPerGpu = (gpuClass, model) => {
    const pair = facts.byClassModel.get(gpuClass + "|" + model);
    if (pair && pair.gpus >= MIN_SAMPLE_GPUS) return { w: median(pair.perGpu), ...spread(pair), gpus: pair.gpus, basis: "model" };
    const cls = facts.byClass.get(gpuClass);
    if (cls && cls.gpus > 0) return { w: median(cls.perGpu), ...spread(cls), gpus: cls.gpus, basis: "class" };
    return null;
  };

  const running = new Set(facts.runningModels);
  const skipped = new Set();
  const seen = new Set();
  const configs = [];
  // Setups Gonka publishes for a kind of GPU no host runs now: nothing to
  // measure, so they are named rather than priced.
  const noHosts = [];
  // Machines running exactly this rig now, and the hosts they belong to.
  const runningNow = (gpuClass, model, gpuCount) => {
    const s = facts.sizes.get(`${gpuClass}|${model}|${gpuCount}`);
    return { machines: s ? s.machines : 0, hosts: s ? s.hosts.size : 0 };
  };
  const rig = (row, per, gpuCount, copies) => {
    const gnkPerDay = per.w * gpuCount * facts.gnkPerWeight * perDay;
    return {
      model: row.model,
      gpuClass: row.gpuClass,
      gpuCount,
      // How many copies of the setup the ML node runs on this many GPUs.
      copies,
      ...runningNow(row.gpuClass, row.model, gpuCount),
      weight: Math.round(per.w * gpuCount),
      gnkPerDay,
      // Where half the hosts running it fall, in GNK a day at this rig's size (see spread above).
      // Only where this exact model is measured on this kind of GPU: a figure borrowed from the
      // kind's other models (basis "class") mixes models, and its range landed below the figure.
      // The range reaches the figure itself when the biggest hosts sit at one end (8 x B300, Nov 2026).
      range: per.basis === "model" ? hug(rangeOf(per.lo, per.hi, per.n, gpuCount), per.w * gpuCount * facts.gnkPerWeight * perDay) : null,
      // Every host behind the figure (over-reporters too): the page says so when it is too few for a range.
      measuredHosts: per.all,
      usdPerDay: price.usd ? gnkPerDay * price.usd : null,
      // What the whole rig may cost per hour before it stops paying for itself.
      breakEvenPerHour: price.usd ? (gnkPerDay * price.usd) / 24 : null,
      basis: per.basis,          // "model" = GPUs of this class running this model; "class" = all of this class
      sampleGpus: Math.round(per.gpus),
      deployable: true
    };
  };
  for (const row of rows) {
    if (!row.gpuClass || !row.model || !row.gpuCount) continue;
    if (running.size && !running.has(row.model)) { skipped.add(row.model); continue; }
    const key = `${row.model}|${row.gpuClass}|${row.gpuCount}`;
    if (seen.has(key)) continue;
    seen.add(key);
    const per = weightPerGpu(row.gpuClass, row.model);
    // The same setup on a whole server too, when it divides evenly.
    const whole = row.gpuCount < FULL_SERVER && FULL_SERVER % row.gpuCount === 0 && !seen.has(`${row.model}|${row.gpuClass}|${FULL_SERVER}`);
    if (whole) seen.add(`${row.model}|${row.gpuClass}|${FULL_SERVER}`);
    if (!per) {
      const none = { model: row.model, gpuClass: row.gpuClass, machines: 0, hosts: 0 };
      noHosts.push({ ...none, gpuCount: row.gpuCount, copies: 1 });
      if (whole) noHosts.push({ ...none, gpuCount: FULL_SERVER, copies: FULL_SERVER / row.gpuCount });
      continue;
    }
    configs.push(rig(row, per, row.gpuCount, 1));
    if (whole) configs.push(rig(row, per, FULL_SERVER, FULL_SERVER / row.gpuCount));
  }

  // Models the network pays for that no published configuration covers — GLM
  // is one today. People do run them, with a node-config they wrote themselves,
  // so leaving them out of the table looks like the app is hiding something.
  // They are listed exactly as the network shows them, marked not deployable.
  // Hosts declare their own hardware, so a kind only one host reports is left
  // out: one host's say-so put an RTX 4090 above an H100 (Oct 2026).
  for (const [key, v] of facts.byClassModel) {
    const [gpuClass, model] = key.split("|");
    if (!running.has(model) || v.gpus < MIN_OBSERVED_GPUS || v.hosts.size < MIN_OBSERVED_HOSTS) continue;
    if (configs.some((c) => c.model === model && c.gpuClass === gpuClass)) continue;
    const gpuCount = Math.round(median(v.nodeGpus)) || 1;
    const w = median(v.perGpu);
    const gnkPerDay = w * gpuCount * facts.gnkPerWeight * perDay;
    configs.push({
      model, gpuClass, gpuCount,
      ...runningNow(gpuClass, model, gpuCount),
      weight: Math.round(w * gpuCount),
      gnkPerDay,
      range: (({ lo, hi, n }) => hug(rangeOf(lo, hi, n, gpuCount), gnkPerDay))(spread(v)),
      measuredHosts: spread(v).all,
      usdPerDay: price.usd ? gnkPerDay * price.usd : null,
      breakEvenPerHour: price.usd ? (gnkPerDay * price.usd) / 24 : null,
      basis: "model",
      sampleGpus: Math.round(v.gpus),
      deployable: false
    });
  }
  configs.sort((a, b) => b.gnkPerDay - a.gnkPerDay);

  const classes = [...facts.byClass.entries()].filter(([, v]) => v.hosts.size >= MIN_OBSERVED_HOSTS).map(([id, v]) => ({
    id, gpus: v.gpus, nodes: v.perGpu.length,
    gnkPerGpuPerDay: median(v.perGpu) * facts.gnkPerWeight * perDay
  })).sort((a, b) => b.gnkPerGpuPerDay - a.gnkPerGpuPerDay);

  const data = {
    asOf: new Date().toISOString(),
    epoch: {
      index: facts.epochIndex,
      hours: facts.epochHours,
      mintedGnk: facts.mintedGnk,
      // What the hosts were paid, and in which epoch: the amount shared out.
      // "minted" when the payout record could not be read (then the figures
      // run high, by as much as the part that does not reach the hosts).
      basis: typical ? "typical" : paid ? "paid" : "minted",
      // The part of its full share a working host was paid, and from which epochs.
      typicalShare: typical ? typical.share : null,
      typicalEpochs: typical ? typical.epochs : null,
      paidGnk: paid ? paid.gnk : null,
      paidEpoch: paid ? paid.epoch : null,
      paidHosts: paid ? paid.hosts : null,
      paidShare: paid ? paid.share : null,
      totalWeight: facts.totalWeight,
      participants: facts.participants,
      gnkPerWeight: facts.gnkPerWeight,
      networkGnkPerDay: (paid ? paid.gnk : facts.mintedGnk) * perDay
    },
    price,
    vesting: {
      epochs: facts.vestingEpochs,
      days: facts.vestingEpochs * facts.epochHours / 24
    },
    classes,
    configs,
    // Configurations the repo offers for models the network is not serving this
    // epoch: no observed weight, so they are left out rather than guessed at.
    withoutData: [...skipped].sort(),
    // Setups Gonka publishes that no host runs now, so nothing can be measured.
    noHosts,
    // Machines left out because their weight cannot come from the GPUs their
    // host reports (more than leftOutAbove × the middle machine of that kind).
    leftOutAbove: LEFT_OUT_ABOVE,
    leftOut: [...facts.leftOut].map(([gpuClass, v]) => ({ gpuClass, machines: v.machines, hosts: v.hosts.size }))
      .sort((a, b) => b.machines - a.machines)
  };
  cache = { at: Date.now(), data };
  return data;
}

const Rentals = (() => {
  const module = { exports: {} };
  const require = need;
/**
 * rentals.js — what renting a server for each setup costs right now, from the
 * providers' own public price lists, next to what that setup earns.
 *
 * Only dedicated (on-demand) servers. A spot, interruptible or preemptible
 * server can be taken back at any moment, and a node stopped partway through
 * an epoch is paid nothing for it, so those are never offered.
 *
 * Each offer is checked against what a node needs before it counts: RAM of at
 * least 1.5 × the GPUs' memory (5% leeway, as the health check allows), the
 * disk Gonka asks for, and a system image with an NVIDIA driver new enough for
 * Gonka's ML node images (CUDA 12.9 or newer; knowledge.requirements). The
 * app works this out live; the website's Base44 function is built from this
 * very code (scripts/build-base44-rentals.js), and the data job publishes the
 * same as a fallback (scripts/rentals-snapshot.js). Same code, same numbers.
 *
 * Spheron first: its documented GET /api/gpu-offers needs no key
 * (docs.spheron.ai/api-reference), and every price there is for the whole
 * server, per hour. Another provider is one more entry in PROVIDERS, returning
 * offers in the same shape.
 */
const K = require("../knowledge");

/**
 * GPU kinds the network runs, as providers name them: those GNOT can set up,
 * and those hosts run with a node-config of their own (compareOwn).
 * Grace (GH200, GB200/GB300) machines are a different platform.
 */
const KINDS = [[/B300/i, "B300"], [/B200/i, "B200"], [/H200/i, "H200"], [/H100/i, "H100"], [/A100/i, "A100"],
  [/RTX\s*PRO\s*6000/i, "RTX PRO 6000"], [/RTX\s*5090/i, "RTX 5090"], [/\bH20\b/i, "H20"]];
const NOT_THESE = /\bG[BH]\d00\b|\bCC\b/i;
/** Gonka's A100 setup splits the model over 4 × 80 GB, so a 40 GB A100 cannot run it. */
const MIN_GPU_GB = { A100: 75 };
const CACHE_MS = 10 * 60 * 1000;
let cache = null;

async function getJson(url, timeout = 20000) {
  const res = await fetch(url, {
    headers: { "User-Agent": "gonka-network-onboarding-tool (price comparison)", Accept: "application/json" },
    signal: AbortSignal.timeout(timeout)
  });
  if (!res.ok) throw new Error(`HTTP ${res.status} from ${new URL(url).host}`);
  return res.json();
}

const kindOf = (name) => (NOT_THESE.test(name) ? null : (KINDS.find(([re]) => re.test(name)) || [, null])[1]);

/** Is CUDA version a at least b? ("13.0" vs "12.9") */
function cudaAtLeast(a, b) {
  const pa = String(a).split(".").map(Number), pb = String(b).split(".").map(Number);
  for (let i = 0; i < Math.max(pa.length, pb.length); i++) {
    const x = pa[i] || 0, y = pb[i] || 0;
    if (x !== y) return x > y;
  }
  return true;
}
/** The newest CUDA version named in a list of system images, or null. */
function newestCuda(images) {
  let best = null;
  for (const name of images) {
    // "CUDA 13.0" and "CUDA 13" alike: Spheron names its B300 images
    // "Ubuntu 24.04 (CUDA 13)", which a version with a dot never matched.
    const m = /CUDA\s*(\d+)(?:\.(\d+))?/i.exec(name);
    if (!m) continue;
    const v = `${m[1]}.${m[2] || "0"}`;
    if (!best || cudaAtLeast(v, best)) best = v;
  }
  return best;
}

/** Spheron's dedicated offers, in the shared shape. */
async function spheron() {
  const j = await getJson("https://app.spheron.ai/api/gpu-offers?page=1&limit=200&computeType=gpu");
  const out = [];
  for (const g of j.data || []) {
    const gpu = kindOf(`${g.displayName || ""} ${g.gpuType || ""}`);
    if (!gpu) continue;
    for (const o of g.offers || []) {
      if (!o.available || o.instanceType !== "DEDICATED") continue;
      const gpuGB = Number(o.gpu_memory) || 0;
      if (MIN_GPU_GB[gpu] && gpuGB < MIN_GPU_GB[gpu]) continue;
      out.push({
        provider: "Spheron",
        gpu, gpuName: g.displayName || gpu,
        gpus: Number(o.gpuCount) || 0,
        gpuGB,
        ramGB: Number(o.memory) || 0,
        diskGB: Number(o.storage) || 0,
        cpus: Number(o.vcpus) || 0,
        // Some regions carry an internal prefix: "ce32df2f:tokyo-japan-5".
        region: String((o.clusters || [])[0] || "").replace(/^[0-9a-f]{6,}:/i, ""),
        pricePerHour: Number(o.price) || 0,
        // The newest CUDA among the images on offer ("Ubuntu 24.04 LTS (CUDA 13.0)",
        // "Ubuntu Server 22.04 LTS R570 CUDA 12.8"), or null when none comes with it.
        cuda: newestCuda(o.os_options || [])
      });
    }
  }
  return out.filter((o) => o.gpus > 0 && o.pricePerHour > 0);
}

const PROVIDERS = [{ id: "spheron", name: "Spheron", site: "https://www.spheron.network/", read: spheron }];

/** What an offer lacks for a node: "memory", "disk" and/or "driver". */
function shortOf(o) {
  const req = K.get().requirements;
  const short = [];
  const needRam = o.gpus * o.gpuGB * req.mlNodeRamToVramRatio;
  if (o.ramGB < needRam * 0.95) short.push("memory");
  if (o.diskGB < req.networkNode.diskGB) short.push("disk");
  // Gonka's images need CUDA 13, or 12.9 for the fallback one (knowledge.requirements).
  if (!o.cuda || !cudaAtLeast(o.cuda, req.cudaMin)) short.push("driver");
  return short;
}

/**
 * One row per server a GNOT setup can run on: a GPU kind and a GPU count that
 * is a whole multiple of a setup's (one copy per set of GPUs). Each row has the
 * cheapest dedicated offer that has everything a node needs, or else the
 * cheapest offer and what it lacks, and what the server earns with each model
 * it can run, from the earnings figures (`earnings` is earnings.estimate()).
 */
function compare(earnings, offers) {
  // Per GPU kind and model: the setup's GPU count, and per GPU what it mines a
  // day, what that is worth and what it may cost per hour.
  const perGpu = (v, n) => (v != null ? v / n : null);
  const setups = new Map();
  for (const c of earnings.configs || []) {
    if (c.deployable === false || !(c.copies <= 1)) continue;
    setups.set(`${c.gpuClass}|${c.model}`, { gpuClass: c.gpuClass, model: c.model, per: c.gpuCount,
      perGpuHour: perGpu(c.breakEvenPerHour, c.gpuCount), perGpuGnk: perGpu(c.gnkPerDay, c.gpuCount), perGpuUsd: perGpu(c.usdPerDay, c.gpuCount),
      // What the middle half of hosts running it earn (earnings.js range), per GPU in GNK a day.
      perGpuLow: c.range ? c.range.low / c.gpuCount : null, perGpuHigh: c.range ? c.range.high / c.gpuCount : null });
  }
  const gnkUsd = earnings.price && earnings.price.usd ? earnings.price.usd : null;
  // What is left each day after the rent, for the middle half of hosts: { low, high } in dollars.
  const gapRange = (range, best) => (range && best && gnkUsd
    ? { low: range.low * gnkUsd - best.pricePerHour * 24, high: range.high * gnkUsd - best.pricePerHour * 24 } : null);
  for (const c of earnings.noHosts || []) {
    if (c.copies > 1 || setups.has(`${c.gpuClass}|${c.model}`)) continue;
    setups.set(`${c.gpuClass}|${c.model}`, { gpuClass: c.gpuClass, model: c.model, per: c.gpuCount, perGpuHour: null, perGpuGnk: null, perGpuUsd: null });
  }
  // The servers worth a row: each setup's own size and a whole 8-GPU server,
  // plus any size on offer that is a whole multiple of a setup's.
  const sizes = new Map();
  const want = (gpu, n) => { if (!sizes.has(gpu)) sizes.set(gpu, new Set()); sizes.get(gpu).add(n); };
  for (const s of setups.values()) {
    want(s.gpuClass, s.per);
    if (s.per < 8 && 8 % s.per === 0) want(s.gpuClass, 8);
  }
  for (const o of offers) for (const s of setups.values()) if (o.gpu === s.gpuClass && o.gpus % s.per === 0) want(o.gpu, o.gpus);

  const rows = [];
  for (const [gpu, counts] of sizes) for (const n of counts) {
    const runs = [...setups.values()].filter((s) => s.gpuClass === gpu && n % s.per === 0)
      .map((s) => ({ model: s.model, copies: n / s.per, breakEvenPerHour: s.perGpuHour != null ? s.perGpuHour * n : null,
        gnkPerDay: s.perGpuGnk != null ? s.perGpuGnk * n : null, usdPerDay: s.perGpuUsd != null ? s.perGpuUsd * n : null,
        range: s.perGpuLow != null ? { low: s.perGpuLow * n, high: s.perGpuHigh * n } : null }))
      .sort((a, b) => (b.breakEvenPerHour || 0) - (a.breakEvenPerHour || 0));
    if (!runs.length) continue;
    const here = offers.filter((o) => o.gpu === gpu && o.gpus === n).map((o) => ({ ...o, short: shortOf(o) }))
      .sort((a, b) => a.pricePerHour - b.pricePerHour);
    const best = here.find((o) => !o.short.length) || null;
    const top = runs[0].breakEvenPerHour;
    rows.push({
      gpuClass: gpu, gpuCount: n, runs,
      offers: here.length,
      best,
      // Nothing with all a node needs: the cheapest anyway, with what it lacks.
      cheapest: best ? null : here[0] || null,
      // What the best-earning model leaves after the rent, per day.
      gapPerDay: best && top != null ? (top - best.pricePerHour) * 24 : null,
      gapRange: gapRange(runs[0].range, best)
    });
  }
  // Cheapest first: what can be rented as it is, then offers lacking something,
  // then sizes nobody offers, each from the cheapest to the most expensive.
  const order = ["B300", "B200", "H200", "H100", "A100"];
  const rank = (r) => (r.best ? 0 : r.cheapest ? 1 : 2);
  const price = (r) => (r.best || r.cheapest || { pricePerHour: Infinity }).pricePerHour;
  rows.sort((a, b) => rank(a) - rank(b) || price(a) - price(b)
    || order.indexOf(a.gpuClass) - order.indexOf(b.gpuClass) || a.gpuCount - b.gpuCount);
  return rows;
}

/**
 * The rigs the network runs and pays for that have no ready-made configuration
 * (marked † in the table): possible on Gonka, so they are compared too, each
 * with its own row, after the rigs GNOT can set up.
 */
function compareOwn(earnings, offers) {
  const price = earnings.price && earnings.price.usd ? earnings.price.usd : null;
  return (earnings.configs || []).filter((c) => c.deployable === false).map((c) => {
    const here = offers.filter((o) => o.gpu === c.gpuClass && o.gpus === c.gpuCount).map((o) => ({ ...o, short: shortOf(o) }))
      .sort((a, b) => a.pricePerHour - b.pricePerHour);
    const best = here.find((o) => !o.short.length) || null;
    return {
      gpuClass: c.gpuClass, gpuCount: c.gpuCount, own: true,
      runs: [{ model: c.model, copies: c.copies, breakEvenPerHour: c.breakEvenPerHour, gnkPerDay: c.gnkPerDay, usdPerDay: c.usdPerDay, range: c.range ? { low: c.range.low, high: c.range.high } : null }],
      offers: here.length,
      best,
      cheapest: best ? null : here[0] || null,
      gapPerDay: best && c.breakEvenPerHour != null ? (c.breakEvenPerHour - best.pricePerHour) * 24 : null,
      gapRange: c.range && best && price ? { low: c.range.low * price - best.pricePerHour * 24, high: c.range.high * price - best.pricePerHour * 24 } : null
    };
  });
}

/**
 * A provider's offers, read again when the list comes back much shorter than
 * it was: on 26 October 2026 the website read Spheron's list with a single
 * offer GNOT could use, where there are usually dozens, and five minutes later
 * every one was back. A short list (under 10 usable offers, or under half of
 * the last read) is read up to twice more, a few seconds apart, and the
 * fullest read counts. A list that stays short is shown as it is.
 */
const lastCount = {};
const THIN_RETRIES = 2;
const THIN_WAIT_MS = 4000;
async function readFull(p) {
  const thin = (list) => list.length < Math.max(10, (lastCount[p.id] || 0) / 2);
  let list = await p.read();
  for (let i = 0; i < THIN_RETRIES && thin(list); i++) {
    await new Promise((r) => setTimeout(r, THIN_WAIT_MS));
    const again = await p.read().catch(() => []);
    if (again.length > list.length) list = again;
  }
  lastCount[p.id] = list.length;
  return list;
}

/**
 * Every provider's dedicated offers, read now, each compared on its own with
 * the earnings figures: one list per provider, so people can look at them one
 * provider at a time.
 */
async function estimate(earnings, { force = false } = {}) {
  if (!force && cache && Date.now() - cache.at < CACHE_MS && cache.epoch === (earnings.epoch && earnings.epoch.index)) return cache.data;
  const providers = [];
  for (const p of PROVIDERS) {
    const at = new Date().toISOString();
    try {
      const list = await readFull(p);
      providers.push({ id: p.id, name: p.name, site: p.site, at, offers: list.length, error: null, rows: compare(earnings, list), ownRows: compareOwn(earnings, list) });
    } catch (e) {
      providers.push({ id: p.id, name: p.name, site: p.site, at, offers: 0, error: e.message, rows: [], ownRows: [] });
    }
  }
  const data = { asOf: new Date().toISOString(), providers };
  cache = { at: Date.now(), epoch: earnings.epoch && earnings.epoch.index, data };
  return data;
}

// The data job's last earnings.json: the price of each market that does not

let published = { at: 0, f: null };

    try { published = { at: Date.now(), f: await (await fetch(EARNINGS_URL, { signal: AbortSignal.timeout(15000) })).json() }; }

// The markets the app reads, in its order (src/services/earnings.js gnkPrice).
const MARKETS = ["CoinGecko","CoinPaprika"];
function fillInPrice(live, published) {
  const have = new Map(((live && live.sources) || []).map((s) => [s.name, s]));
  const answered = have.size;
  for (const s of (published && published.sources) || []) {
    if (!have.has(s.name) && s.usd > 0) have.set(s.name, { name: s.name, usd: s.usd, readAt: published.asOf });
  }
  const sources = [...have.values()];
  if (!sources.length) return { usd: null, sources: [] };
  const sorted = sources.map((s) => s.usd).sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  const usd = sorted.length % 2 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
  return answered ? { usd, sources } : { usd, sources, published: published && published.asOf };
}
function marketsFrom(body) {
  const seen = new Map();
  for (const m of (body && Array.isArray(body.markets) ? body.markets : [])) {
    const usd = Number(m && m.usd);
    if (m && MARKETS.includes(m.name) && usd > 0 && isFinite(usd) && !seen.has(m.name)) seen.set(m.name, { name: m.name, usd });
  }
  return [...seen.values()];
}
function priceFrom(browser, own, published) {
  const have = new Map(browser.map((s) => [s.name, s]));
  for (const s of (own && own.sources) || []) if (!have.has(s.name)) have.set(s.name, { name: s.name, usd: s.usd });
  const p = fillInPrice({ sources: [...have.values()] }, published);

async function getJson(url, timeoutMs = 10000) {
  const res = await fetch(url, { signal: AbortSignal.timeout(timeoutMs) });
  if (res.status === 404) return null;                    // "nothing recorded" is an answer, not a failure
  if (res.status === 503 || res.status === 429) { const e = new Error("the node asked us to slow down"); e.slowDown = true; throw e; }
  // Cloudflare's own answers when the node behind it is down (520-530; node3 since Nov 9, 2026, as the
  // website's server sees it): the node cannot be reached, nothing was refused. Counted as a refusal, it
  // left a question one refusal from being given up on.
  if (res.status >= 520 && res.status <= 530) throw new Error(`node down (Cloudflare ${res.status}) at ${url}`);
  if (!res.ok) throw new Error(`HTTP ${res.status} from ${url}`);

/**
 * Ask Gonka's public nodes, at a pace they accept.
 *
 * Measured in October 2026: a node answers about twenty questions in a burst
 * from one address and then refuses (HTTP 503) until it has had a rest, while
 * one every 0.7 seconds goes on for ever. A screen that fired its fifty
 * questions together got refusals and took half a minute. So every question
 * waits for a turn: each node has an allowance that starts at ALLOWANCE and
 * refills at PER_SECOND, and a refusal is answered by waiting, not by asking
 * again at once.
 *
 * The nodes are not equal either: node2 answered most questions but failed
 * (HTTP 500) on every host's past results and on the reward estimate, node1
 * answered everything, node3 did not answer at all. So each question goes to
 * whichever node has the most allowance left among those not known to fail
 * that kind of question, which spreads a read over the nodes; a node that
 * fails a kind is left alone for that kind for a while, one that cannot be
 * reached is left alone altogether, and a merely slow one is joined by the
 * next after a moment.
 */
const ALLOWANCE = 15;
const PER_SECOND = 1.3;
const SLOW_MS = 2500;
const AVOID_KIND_MS = 10 * 60 * 1000;
const AVOID_NODE_MS = 3 * 60 * 1000;
const pace = {};                 // node -> { left, at }
const failed = {};               // "node|kind" or "node|*" -> until when to leave it alone
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
/** The last questions that went wrong, oldest first: what the website's server can say about itself. */
const trouble = [];
function noteTrouble(node, path, e) {
  trouble.push({ at: new Date().toISOString(), node, path: String(path).split("?")[0], error: String((e && e.message) || e).replace(/ from https?:\/\/\S+/, "").slice(0, 200) });
  if (trouble.length > 40) trouble.shift();
}
function recentTrouble() {
  const now = Date.now();
  return {
    trouble: trouble.slice(),
    leftAlone: Object.entries(failed).filter(([, until]) => until > now).map(([what, until]) => ({ what, forSeconds: Math.round((until - now) / 1000) })),
    allowance: Object.fromEntries(Object.keys(pace).map((n) => [n, Math.round(allowance(n).left * 10) / 10]))
  };
}
function allowance(node) {
  const p = pace[node] || (pace[node] = { left: ALLOWANCE, at: Date.now() });
  const now = Date.now();
  p.left = Math.min(ALLOWANCE, p.left + (now - p.at) / 1000 * PER_SECOND);
  p.at = now;
  return p;
}
async function turn(node) {
  for (;;) {
    const p = allowance(node);
    if (p.left >= 1) { p.left -= 1; return; }
    await sleep(Math.ceil((1 - p.left) / PER_SECOND * 1000));
  }
}
async function paced(node, path, timeoutMs) {
  for (let attempt = 0; ; attempt++) {
    await turn(node);
    try { return await getJson(node + path, timeoutMs); }
    catch (e) {
      if (e.slowDown) noteTrouble(node, path, new Error(`asked to slow down (try ${attempt + 1})`));
      if (!e.slowDown || attempt >= 2) throw e;
      allowance(node).left = 0;                          // told to slow down: start the allowance again
      await sleep(2000 * (attempt + 1));
    }
  }
}
function ask(seeds, path, timeoutMs) {
  const kind = path.replace(/^\/chain-api\/productscience\/inference\//, "").split("/").slice(0, 2).join("/").replace(/\?.*$/, "");
  const now = Date.now();
  const avoid = (n) => (failed[n + "|" + kind] || 0) > now || (failed[n + "|*"] || 0) > now;
  const fit = seeds.filter((n) => !avoid(n));
  // Most allowance first; the ones to avoid stay as a last resort.
  const order = [...fit.sort((a, b) => allowance(b).left - allowance(a).left), ...seeds.filter(avoid)];
  return new Promise((resolve, reject) => {
    let done = false, pending = 0, next = 0, last = null, timer = null, refusals = 0;
    const start = () => {
      if (done || next >= order.length) return;
      const node = order[next++];
      pending++;
      paced(node, path, timeoutMs).then((answer) => {
        pending--;
        if (done) return;
        done = true; clearTimeout(timer);
        resolve(answer);
      }, (e) => {
        pending--; last = e;
        noteTrouble(node, path, e);
        // An HTTP error is about this kind of question; anything else means the node cannot be reached.
        failed[node + "|" + (/^HTTP \d/.test(String(e.message)) ? kind : "*")] = Date.now() + (/^HTTP \d/.test(String(e.message)) ? AVOID_KIND_MS : AVOID_NODE_MS);
        if (done) return;
        // Two nodes refusing the same question is the answer: the question itself
        // is refused (since mid-October 2026 the nodes cap what one may cost).
        if (/^HTTP \d/.test(String(e.message)) && ++refusals >= 2) { done = true; clearTimeout(timer); reject(e); return; }
        if (next < order.length) { clearTimeout(timer); start(); }
        else if (pending === 0) { done = true; reject(last || new Error("no Gonka node answered")); }
      });
      if (next < order.length) timer = setTimeout(start, SLOW_MS);
    };
    if (!order.length) reject(new Error("no Gonka node to ask"));
    else start();
  });
}

/**
 * Who chose what for the models they do not run, across the epoch's hosts.
 *
 * Gonka asks every host, for each model it does not run, to delegate it (a
 * host that runs it checks the work for this one, for a share of its reward),
 * or to refuse it; no choice costs more. Read from the chain's own rules:
 * params.delegation_params gives the rates (in November 2026: delegate 5%,
 * refuse 10%, nothing 15%, per model, added up), params.poc_params.models the
 * models that count. A delegation only counts while the chosen host runs that
 * model and is in the epoch; otherwise the chain treats it as no choice
 * (inference-chain/x/inference/module/delegation_weight_calculator.go).
 *
 * One question per host (about thirty), at the paced rate: kept two minutes.
 */
let choicesCache = { at: 0, data: null };
async function modelChoices(seeds) {
  if (choicesCache.data && Date.now() - choicesCache.at < 2 * 60 * 1000) return choicesCache.data;
  const [m, params] = await Promise.all([members(seeds), ask(seeds, `${INF}/inference/params`)]);
  const p = (params && params.params) || {};
  const dec = (d) => (d && d.value !== undefined ? num(d.value) * Math.pow(10, num(d.exponent)) : null);
  const dp = p.delegation_params || {};
  const rates = { delegate: dec(dp.delegation_share), refuse: dec(dp.refusal_penalty), none: dec(dp.no_participation_penalty) };
  const counted = ((p.poc_params || {}).models || []).map((x) => ({ id: x.model_id, penaltyStart: num(x.penalty_start_epoch) }))
    .filter((x) => x.id && (!x.penaltyStart || x.penaltyStart <= m.epoch + 1));
  // Hosts on Gonka's own lists (developer access, transfer agents) are never
  // GNOT's default pick: Gonka asks hosts not to all lean on the same few.
  const listed = new Set([...(((p.developer_access_params || {}).allowed_developer_addresses) || []),
    ...(((p.transfer_agent_access_params || {}).allowed_transfer_addresses) || [])]);
  const hosts = [];
  for (const h of m.hosts) {
    const d = await soft(ask(seeds, `${INF}/inference/poc_delegation/${h.address}`));
    const choices = {};
    for (const x of ((d && d.delegations) || [])) choices[x.model_id] = { mode: "delegate", to: x.delegate_to };
    for (const x of ((d && d.refusals) || [])) choices[x.model_id] = { mode: "refuse" };
    hosts.push({ address: h.address, weight: h.weight, models: h.models, choices, read: !!d, listed: listed.has(h.address) });
  }
  const total = hosts.reduce((a, h) => a + h.weight, 0);
  const byModel = {};
  for (const model of counted.map((c) => c.id)) {
    byModel[model] = hosts.filter((h) => h.models.includes(model)).map((h) => {
      const from = hosts.filter((o) => o.address !== h.address && o.choices[model] && o.choices[model].mode === "delegate" && o.choices[model].to === h.address);
      const w = from.reduce((a, o) => a + o.weight, 0);
      return { address: h.address, weight: h.weight, delegatedBy: from.length, delegatedWeight: w, delegatedShare: total ? w / total : 0, listed: h.listed };
    }).sort((a, b) => b.weight - a.weight);
  }
  const data = { epoch: m.epoch, total, rates, counted, hosts, byModel, at: Date.now() };
  choicesCache = { at: Date.now(), data };
  return data;
}

}
/** epoch -> when it began (Unix seconds), for the epochs the file has dated. */
function epochStarts(file) {
  const out = {};
  for (const e of (file && file.epochs) || []) if (e.t) out[e.e] = num(e.t);
  return out;
}
/**
 * When an epoch began, as an ISO time, and whether that is the chain's own
 * record (`est: false`) or worked out from the current epoch and the pace of
 * blocks, for an epoch the file has not dated yet.
 */
function epochWhen(e, now) {
  const t = archive.began[e];
  if (t) return { at: new Date(t * 1000).toISOString(), est: false };
  if (!now || !now.secPerBlock || !now.epochBlocks) return { at: null, est: true };
  const startNow = Date.now() - Math.max(0, now.height - now.pocStart) * now.secPerBlock * 1000;
  return { at: new Date(startNow - (now.index - e) * now.epochBlocks * now.secPerBlock * 1000).toISOString(), est: true };
}
/** The file's compact rows back into the rows the screen works from. */
function unpackEpochs(file) {
  const hosts = (file && file.hosts) || [];
  const out = {};
  for (const e of (file && file.epochs) || []) {
    out[e.e] = (e.r || []).map((r) => ({
      address: hosts[r[0]], rewarded: num(r[1]), earned: num(r[2]), requests: num(r[3]), missed: num(r[4]),
      validated: num(r[5]), invalidated: num(r[6]), claimed: !!r[7], burned: 0
    })).filter((r) => r.address);
  }
  return out;
}

/**
 * The last `count` finished epochs, oldest first: [{ epoch, rows }], with rows
 * null for an epoch the prepared file does not have yet.
 *
 * Only from the prepared file. The chain used to answer "every host's results
 * for this epoch" in one question; from about 13 October 2026 the public nodes
 * cap what a question may cost and refuse that one, and asking each host in
 * turn is thirty questions an epoch — fine for the data job, too many for a
 * screen. The job runs several times a day, so the newest epoch can be missing
 * for a few hours; it shows as "not collected yet".
 */
async function history(seeds, currentEpoch, count) {
  const kept = (await epochArchive()) || {};
  const out = [];
  for (let e = Math.max(1, currentEpoch - count); e <= currentEpoch - 1; e++) out.push({ epoch: e, rows: kept[e] || null });
  return out;
}

/** Everything the chain holds about one address. Each part may be missing. */
async function host(seeds, address) {
  const q = (p) => soft(ask(seeds, p));
  const [record, hardware, delegation, maintenance, credit, collateral, unbonding, balance, allowances, recipients, estimate] = await Promise.all([
    q(`${INF}/inference/participant/${address}`),
    q(`${INF}/inference/hardware_nodes/${address}`),
    q(`${INF}/inference/poc_delegation/${address}`),
    q(`${INF}/inference/maintenance_status/${address}`),
    q(`${INF}/inference/maintenance_credit/${address}`),
    q(`${INF}/collateral/collateral/${address}`),
    q(`${INF}/collateral/unbonding/${address}`),
    q(`/chain-api/cosmos/bank/v1beta1/balances/${address}`),
    q(`/chain-api/cosmos/feegrant/v1beta1/issued/${address}`),
    q(`${INF}/inference/list_claim_recipients/${address}`),
    q(`${INF}/inference/estimate_bitcoin_reward/${address}`)   // refused ("out of gas") since mid-October 2026; shown only if it answers
  ]);
  // Rewards can be paid to another address; that is where the vesting lives.
  const entries = ((recipients && recipients.entries) || []).slice().sort((a, b) => num(a.epoch) - num(b.epoch));
  const recipient = entries.length ? entries[entries.length - 1].recipient : address;
  const [schedule, recipientBalance] = await Promise.all([
    q(`${INF}/streamvesting/vesting_schedule/${recipient}`),
    recipient === address ? Promise.resolve(balance) : q(`/chain-api/cosmos/bank/v1beta1/balances/${recipient}`)
  ]);
  return { address, record, hardware, delegation, maintenance, credit, collateral, unbonding, balance, allowances, recipient, schedule, recipientBalance, estimate };
}

const median = (list) => {
  const a = list.filter((x) => isFinite(x)).sort((x, y) => x - y);
  if (!a.length) return null;
  const m = Math.floor(a.length / 2);
  return a.length % 2 ? a[m] : (a[m - 1] + a[m]) / 2;
};
/** 1 = best. Ties share the better place. null when the address is not in the list. */
function place(rows, key, address) {
  const mine = rows.find((r) => r.address === address);
  if (!mine) return null;
  return { rank: rows.filter((r) => r[key] > mine[key]).length + 1, of: rows.length };
}
const share = (part, whole) => (whole > 0 ? part / whole : 0);
/** Of the requests a host was given, the part it did not answer. */
const missedShare = (r) => share(r.missed, r.requests + r.missed);
/** Of its answers that were checked, the part judged wrong. */
const invalidShare = (r) => share(r.invalidated, r.validated + r.invalidated);

/** One epoch's results for one host, with where it stood among the others. */
function epochLine(epoch, rows, address) {
  if (!rows) return { epoch, took: false, unknown: true, hosts: 0 };
  const mine = rows.find((r) => r.address === address);
  if (!mine) return { epoch, took: false, hosts: rows.length };
  const total = rows.reduce((a, r) => a + r.rewarded, 0);
  return {
    epoch, took: true, hosts: rows.length,
    reward: mine.rewarded, work: mine.earned, claimed: mine.claimed,
    requests: mine.requests, missed: mine.missed, validated: mine.validated, invalidated: mine.invalidated,
    // Some epochs carry no request counts for any host at all (434, 438 and
    // 440 in October 2026): not recorded, which is not the same as none answered.
    requestsRecorded: rows.some((r) => r.requests + r.missed > 0),
    missedShare: missedShare(mine), invalidShare: invalidShare(mine),
    rewardShare: share(mine.rewarded, total),
    rewardPlace: place(rows, "rewarded", address),
    typicalMissedShare: median(rows.map(missedShare)),
    typicalInvalidShare: median(rows.filter((r) => r.validated + r.invalidated > 0).map(invalidShare)),
    typicalReward: median(rows.map((r) => r.rewarded))
  };
}

/**
 * A participant's join_time as an ISO date. Hosts that joined later give
 * milliseconds; the ones there from the network's start (join_height 0) give
 * seconds (1755848273, 22 August 2025), which read as milliseconds land in
 * January 1970.
 */
const joinedAt = (v) => new Date(v < 1e11 ? v * 1000 : v).toISOString();

function summarise({ address, now, mem, hist, raw }) {
  const me = mem.hosts.find((h) => h.address === address) || null;
  const out = mem.excluded.find((x) => x.address === address) || null;
  const totalWeight = mem.hosts.reduce((a, h) => a + h.weight, 0);
  const rec = (raw.record && raw.record.participant) || null;
  const stats = (rec && rec.current_epoch_stats) || {};
  const dec = (d) => (d && d.value !== undefined ? num(d.value) * Math.pow(10, num(d.exponent)) : null);

  const lines = hist.map((h) => ({ ...epochLine(h.epoch, h.rows, address), ...epochWhen(h.epoch, now) }));
  const took = lines.filter((l) => l.took);
  const last = took.length ? took[took.length - 1] : null;

  const nodes = (((raw.hardware || {}).nodes || {}).hardware_nodes || []).map((n) => ({
    id: n.local_id, status: n.status || "", models: n.models || [], version: n.version || "",
    gpus: (n.hardware || []).map((g) => ({ type: String(g.type || "").split("|")[0].trim(), memory: (String(g.type || "").split("|")[1] || "").trim(), count: num(g.count) })),
    pocWeight: me ? ((me.nodes.find((x) => x.id === n.local_id) || {}).pocWeight || 0) : 0
  })).map((n) => {
    // What counts for the epoch: what it proved times its model's rate.
    const rate = n.models.length && mem.rates ? mem.rates[n.models[0]] : undefined;
    return { ...n, rate: rate != null ? rate : null, counts: rate != null && n.pocWeight ? Math.floor(n.pocWeight * rate) : null };
  });
  const gpuCount = nodes.reduce((a, n) => a + n.gpus.reduce((b, g) => b + g.count, 0), 0);
  const gpuKinds = {};
  for (const n of nodes) for (const g of n.gpus) gpuKinds[g.type] = (gpuKinds[g.type] || 0) + g.count;

  // What is left of the allowance the node's own key may spend on fees.
  const allowance = ((raw.allowances && raw.allowances.allowances) || []).map((a) => ({
    grantee: a.grantee,
    left: coins((a.allowance || {}).spend_limit),
    limited: Array.isArray((a.allowance || {}).spend_limit) && (a.allowance || {}).spend_limit.length > 0,
    expires: (a.allowance || {}).expiration || null
  })).sort((a, b) => b.left - a.left)[0] || null;

  const stream = (((raw.schedule || {}).vesting_schedule || {}).epoch_amounts || []).map((e) => coins(e.coins));
  const sum = (a) => a.reduce((x, y) => x + y, 0);

  return {
    address,
    epoch: {
      index: now.index, phase: now.phase, height: now.height,
      blocksToNextPoc: Math.max(0, now.nextPocStart - now.height),
      secondsToNextPoc: Math.max(0, (now.nextPocStart - now.height) * now.secPerBlock),
      progress: now.nextPocStart > now.pocStart ? Math.min(1, Math.max(0, (now.height - now.pocStart) / (now.nextPocStart - now.pocStart))) : 0,
      hours: now.epochBlocks * now.secPerBlock / 3600,
      hosts: mem.hosts.length
    },
    standing: {
      known: !!rec,
      inEpoch: !!me,
      excluded: out ? { reason: out.reason, height: out.height } : null,
      status: rec ? String(rec.status || "") : "",
      weight: me ? me.weight : 0,
      share: me ? share(me.weight, totalWeight) : 0,
      place: me ? place(mem.hosts, "weight", address) : null,
      models: me ? me.models : [],
      epochsCompleted: rec ? num(rec.epochs_completed) : 0,
      joined: rec && num(rec.join_time) ? joinedAt(num(rec.join_time)) : null,
      url: rec ? rec.inference_url || "" : "",
      confirmationRatio: dec(stats.confirmationPoCRatio)
    },
    now: {                                           // this epoch so far, as the node's own record has it
      requests: num(stats.inference_count), missed: num(stats.missed_requests),
      validated: num(stats.validated_inferences), invalidated: num(stats.invalidated_inferences),
      estimate: num((((raw.estimate || {}).settle_amount) || {}).reward_coins)
    },
    last,
    history: lines,
    totals: { epochs: took.length, of: lines.length, unknown: lines.filter((l) => l.unknown).length, reward: sum(took.map((l) => l.reward)), requests: sum(took.map((l) => l.requests)) },
    attention: [],
    hardware: { nodes, gpuCount, gpuKinds, working: nodes.filter((n) => /INFERENCE|POC|POW/i.test(n.status)).length },
    poc: {
      delegations: ((raw.delegation || {}).delegations || []).map((d) => ({ model: d.model_id, to: d.delegate_to })),
      refusals: ((raw.delegation || {}).refusals || []).map((d) => ({ model: d.model_id })),
      intents: ((raw.delegation || {}).intents || []).map((d) => ({ model: d.model_id }))
    },
    maintenance: {
      credit: num((raw.credit || {}).credit_blocks),
      active: !!((raw.maintenance || {}).active_reservation),
      scheduled: (raw.maintenance || {}).scheduled_reservation || null
    },
    money: {
      balance: coins((raw.balance || {}).balances),
      collateral: num(((raw.collateral || {}).amount || {}).amount),
      unbonding: ((raw.unbonding || {}).unbondings || []).map((u) => ({ amount: num((u.amount || {}).amount), epoch: num(u.completion_epoch) })),
      allowance,
      recipient: raw.recipient,
      paidElsewhere: raw.recipient !== address,
      recipientBalance: coins((raw.recipientBalance || {}).balances),
      vesting: { total: sum(stream), epochs: stream.length, next: stream.slice(0, 60), next7: sum(stream.slice(0, 7)), next30: sum(stream.slice(0, 30)) }
    }
  };
}

/**
 * What deserves a look, worst first. Only things the network's own answers
 * show; each is a code and its figures, and the screen writes the sentence.
 * "Clearly more than typical" means over one and a half times the middle host
 * and more than one point above it, so ordinary noise never raises a flag.
 */
const clearlyMore = (mine, typical) => typical !== null && typical !== undefined && mine > typical * 1.5 && mine - typical > 0.01;

/**
 * Where a host's answers missed (or judged wrong) per AI session sit beside the
 * middle host's: "much-more" (twice or more: worth a look), "more" (a quarter
 * more or over), "less" (a fifth less or under) or "same". slack keeps small
 * figures from counting as a difference, and fewer than 5 sessions are never
 * called much more. One rule for a node's card, the hosts list and what is
 * worth a look, so they never disagree.
 */
const WORK_SLACK = { missed: 0.5, invalid: 0.2 };
function beside(mine, typical, slack, sessions) {
  if (mine === null || mine === undefined || typical === null || typical === undefined) return null;
  if (mine - typical > slack && mine >= typical * 2) return sessions >= 5 ? "much-more" : "more";
  if (mine - typical > slack && mine >= typical * 1.25) return "more";
  if (typical - mine > slack && mine <= typical * 0.8) return "less";
  return "same";
}

function attention(v) {
  const out = [];
  const add = (level, code, data = {}) => out.push({ level, code, ...data });
  const st = v.standing, l = v.last, m = v.money;
  if (!st.known) return out;
  if (st.excluded) add("bad", "excluded", { reason: st.excluded.reason });
  else if (!st.inEpoch) add("bad", "not-in-epoch");
  else if (st.answers === false) add("bad", "no-answer", { url: st.url });
  const down = v.hardware.nodes.filter((n) => !/INFERENCE|POC|POW/i.test(n.status));
  if (down.length) add(st.inEpoch ? "warn" : "bad", "machines-down", { machines: down.map((n) => ({ id: n.id, status: n.status })), of: v.hardware.nodes.length });
  if (!m.allowance) add("bad", "no-allowance");
  else if (m.allowance.limited && m.allowance.left < 1e9) add("warn", "allowance-low", { left: m.allowance.left });
  if (m.balance < 1e8) add("warn", "wallet-low", { balance: m.balance });
  if (st.inEpoch && st.confirmationRatio !== null && st.confirmationRatio < 1) add("warn", "confirmation", { ratio: st.confirmationRatio });
  if (l) {
    if (l.reward === 0) add("warn", "no-reward", { epoch: l.epoch });
    if (clearlyMore(l.missedShare, l.typicalMissedShare)) add("warn", "missed", { epoch: l.epoch, mine: l.missedShare, typical: l.typicalMissedShare });
    if (l.validated + l.invalidated > 0 && clearlyMore(l.invalidShare, l.typicalInvalidShare)) add("warn", "judged-wrong", { epoch: l.epoch, mine: l.invalidShare, typical: l.typicalInvalidShare });
  }
  // AI work through devshards: twice or more the middle host's missed or wrong
  // answers per session. A note, listed after anything that stops it earning.
  const w = v.work;
  if (w) {
    if (w.missedVs === "much-more") add("note", "ds-missed", { epoch: w.epoch, missed: w.mine.missed, sessions: w.mine.sessions, mine: w.missedPer, typical: w.typical.missedPer });
    if (w.invalidVs === "much-more") add("note", "ds-wrong", { epoch: w.epoch, invalid: w.mine.invalid, sessions: w.mine.sessions, mine: w.invalidPer, typical: w.typical.invalidPer });
  }
  const rank = { bad: 0, warn: 1, note: 2 };
  return out.sort((a, b) => rank[a.level] - rank[b.level]);
}

/**
 * How a host's AI work went in the newest epoch the data job has its devshard
 * results for (usage.json h: the chain keeps them only until the next epoch
 * ends): AI sessions served, answers missed and judged wrong, and what users
 * paid it, beside the middle host of that epoch, per session so a small host
 * and a big one compare fairly. null when there are none for this address.
 */
function devshardWork(use, address, current) {
  const eps = ((use && use.epochs) || []).filter((u) => u.h && u.h[address]).sort((a, b) => b.e - a.e);
  const u = eps[0];
  if (!u) return null;
  const one = (a) => ({ missed: a[0], invalid: a[1], paid: a[2], sessions: a[5] });
  const all = Object.values(u.h).map(one).filter((x) => x.sessions > 0);
  const per = (x, k) => x[k] / x.sessions;
  const mine = one(u.h[address]);
  const missedPer = mine.sessions ? per(mine, "missed") : null, invalidPer = mine.sessions ? per(mine, "invalid") : null;
  const typical = { sessions: median(all.map((x) => x.sessions)), missedPer: median(all.map((x) => per(x, "missed"))), invalidPer: median(all.map((x) => per(x, "invalid"))), paid: median(all.map((x) => x.paid)) };
  return {
    epoch: u.e, hosts: all.length, mine, missedPer, invalidPer, typical,
    missedVs: beside(missedPer, typical.missedPer, WORK_SLACK.missed, mine.sessions),
    invalidVs: beside(invalidPer, typical.invalidPer, WORK_SLACK.invalid, mine.sessions),
    // The chain adds results as the epoch's AI sessions close, through the next epoch.
    settling: u.e >= current - 1
  };
}

/**
 * Whether a host's public address answers at all. The network learns that a
 * node is gone only when it sends it work and hears nothing back, which for a
 * small node can take hours; its address tells at once. Measured 2026-10-31:
 * all 29 running hosts answered "/" within a third of a second, and a node whose
 * server had been ended an hour before did not answer in 8. Asked twice before
 * saying no: one lost answer is not a node gone. null when there is no public
 * address to ask.
 */
async function answers(url, address) {
  const k = await knock(url, address);
  return k ? k.ok : null;
}

/**
 * What a host's public address answers: { ok, status }, status being the last
 * HTTP status seen (null for none). A node's own front page answers "/" with
 * 200 (Gonka's dashboard). A 5xx is not the node: the website's server reaches
 * the internet through a gateway that answers by itself when a node is not
 * there, and it said every host answered, the ended one too (2026-10-31).
 * null when there is no public address to ask.
 */
async function knock(url, address) {
  const own = await knockHere(url);
  // A 403 says nothing about the node (see settleKnock): ask what the data job saw.
  if (own && own.status === 403 && address) {
    const file = await networkFile();
    return settleKnock(own, file && file.reach, address);
  }
  return own;
}

/**
 * When a knock from here cannot tell, what the data job's knock from GitHub's
 * machines saw (network.json "reach", scripts/network-snapshot.js) decides, if
 * it is less than REACH_MAX_AGE old. A 403 cannot tell: the website's server
 * reaches the internet through a gateway that hands back 403 for most hosts,
 * one that was switched off too (seen 2026-11-09, when 21 of 26 hosts answered
 * it 403 and every live one answered 200 from anywhere else). Without a recent
 * record a 403 still counts as answering, so there is never a false alarm.
 */
const REACH_MAX_AGE = 12 * 3600 * 1000;
function settleKnock(own, reach, address, now = Date.now()) {
  if (!own || own.status !== 403 || !reach || !reach.hosts || !address) return own;
  const age = now - Date.parse(reach.at || "");
  if (!(age >= 0 && age < REACH_MAX_AGE)) return own;
  const seen = reach.hosts[address];
  if (!seen || typeof seen.ok !== "boolean") return own;
  return { ok: seen.ok, status: own.status, seenBy: "data", seenAt: reach.at };
}

/** One knock from this computer: { ok, status }, or null when there is no public address to ask. */
async function knockHere(url) {
  let u;
  try { u = new URL(String(url || "")); } catch (_) { return null; }
  if (!/^https?:$/.test(u.protocol)) return null;
  // Hosts tell their own address: one on a private network is not asked.
  if (/^(localhost|127\.|10\.|192\.168\.|169\.254\.|172\.(1[6-9]|2\d|3[01])\.|0\.|\[)/i.test(u.hostname)) return null;
  let status = null;
  for (let i = 0; i < 2; i++) {
    try {
      const res = await fetch(u.origin + "/", { signal: AbortSignal.timeout(4000) });
      res.arrayBuffer().catch(() => {});
      status = res.status;
      if (status < 500) return { ok: true, status };
    } catch (_) { /* once more */ }
  }
  return { ok: false, status };
}

/** The whole picture for one address. */
async function overview(seeds, address, { epochs = 30 } = {}) {
  const [now, mem, raw, file, use] = await Promise.all([epochNow(seeds), members(seeds), host(seeds, address), networkFile(), usageFile()]);
  const me = mem.hosts.find((h) => h.address === address);
  const [hist, knocked] = await Promise.all([history(seeds, mem.epoch || now.index, epochs), me ? knock(me.url, address) : null]);
  const view = summarise({ address, now, mem, hist, raw });
  view.standing.answers = knocked ? knocked.ok : null;
  // When that was found out: null means just now; a time means the data job's check from GitHub
  // (the website's server cannot reach most hosts itself), which can be hours old.
  view.standing.answersCheckedAt = knocked && knocked.seenAt ? knocked.seenAt : null;
  view.work = devshardWork(use, address, mem.epoch || now.index);
  view.attention = attention(view);
  // Its weight in H100s, and whether it reports far fewer GPUs than that weight
  // takes (scripts/network-snapshot.js): hosts report their own hardware.
  const g = file && file.hosts && file.hosts[address];
  view.hardware.h100 = g && g.h100 != null ? g.h100 : null;
  view.hardware.underReported = !!(g && g.underReported);
  return view;
}

/**
 * Each host's GPUs and the network's totals, as one file kept by the data job
 * (scripts/network-snapshot.js): the GPUs are only told host by host. Kept in
 * memory for ten minutes. null when it cannot be had.
 */
let netFile = { at: 0, data: null };
async function networkFile() {
  const url = K.get().networkUrl;
  if (!url) return null;
  if (netFile.data && Date.now() - netFile.at < 10 * 60 * 1000) return netFile.data;
  try {
    const res = await fetch(url, { signal: AbortSignal.timeout(15000) });
    if (res.ok) netFile = { at: Date.now(), data: await res.json() };
  } catch (_) { /* keep whatever was read before */ }
  return netFile.data;
}

/**
 * What the network's working machines run each model on, and how much memory
 * each kind of GPU has, as network.json carries them (scripts/network-snapshot.js).
 * Host Setup's first step shows them. Null when the file has neither yet.
 */
async function hardwareSetups() {
  const f = await networkFile();
  if (!f || (!f.models && !f.vram)) return null;
  return { asOf: f.asOf || null, models: f.models || {}, vram: f.vram || {} };
}

/**
 * What hosts paid in network fees over about the last epoch (fees.json, from
 * scripts/fees-snapshot.js), summed up for Host Setup's fee card: the hosts
 * with under 1% of the network's weight — one 8-GPU server or less — and the
 * biggest payment of all, in ngonka. Kept ten minutes; null when unknown.
 */
let feeFile = { at: 0, data: null };
async function typicalFees() {
  const url = K.get().feesUrl;
  if (!url) return null;
  if (!feeFile.data || Date.now() - feeFile.at > 10 * 60 * 1000) {
    try {
      const res = await fetch(url, { signal: AbortSignal.timeout(15000) });
      if (res.ok) feeFile = { at: Date.now(), data: await res.json() };
    } catch (_) { /* keep whatever was read before */ }
  }
  const f = feeFile.data;
  const paid = ((f && f.hosts) || []).filter((h) => h.fees != null);
  if (!paid.length) return null;
  const small = paid.filter((h) => h.share < 0.01).map((h) => h.fees).sort((a, b) => a - b);
  return {
    asOf: f.asOf, hours: f.hours, hosts: paid.length,
    small: small.length ? { hosts: small.length, low: small[0], high: small[small.length - 1], typical: median(small) } : null,
    most: Math.max(...paid.map((h) => h.fees))
  };
}

async function capacityFile() {
  const url = K.get().capacityUrl;
  if (!url) return null;
  if (capFile.data && Date.now() - capFile.at < 10 * 60 * 1000) return capFile.data;
  try {
    const res = await fetch(url, { signal: AbortSignal.timeout(15000) });
    if (res.ok) capFile = { at: Date.now(), data: await res.json() };
  } catch (_) { /* keep whatever was read before */ }
  return capFile.data;
}

/**
 * What an epoch minted for its hosts, what they were paid, and why the rest
 * was not, as the chain's reward code works (x/inference/keeper
 * bitcoin_rewards.go): the schedule mints initial × e^(decay × epochs since
 * genesis); each host's share is its weight over ALL the epoch's weight; a
 * host that fails the missed-requests test, or was left out, gets nothing;
 * weight not confirmed during the epoch, and anything over the 30% cap on one
 * host, is cut. What is cut is not shared among the others: it goes to the
 * governance account. `nothing` is the full share of the hosts paid nothing;
 * `rest` is everything else that was not paid. Null when it cannot be worked out.
 */
let rewardParams = { at: 0, p: null };
async function epochPayout(seeds, e, rows) {
  if (!rewardParams.p || Date.now() - rewardParams.at > 60 * 60 * 1000) {
    const j = await ask(seeds, K.get().api.inferenceParams);
    rewardParams = { at: Date.now(), p: j && j.params && j.params.bitcoin_reward_params };
  }
  const bp = rewardParams.p;
  const g = await ask(seeds, `${INF}/inference/epoch_group_data/${e}`);
  const vws = (g && g.epoch_group_data && g.epoch_group_data.validation_weights) || [];
  if (!bp || !bp.use_bitcoin_rewards || !vws.length || !rows.length) return null;
  const decay = num(bp.decay_rate && bp.decay_rate.value) * Math.pow(10, num(bp.decay_rate && bp.decay_rate.exponent));
  const minted = num(bp.initial_epoch_reward) * Math.exp(decay * (e - num(bp.genesis_epoch)));
  const total = vws.reduce((a, v) => a + Math.max(0, num(v.weight)), 0);
  const paid = rows.reduce((a, r) => a + r.rewarded, 0);
  if (!(minted > 0) || !(total > 0) || paid > minted * 1.01) return null;
  let nothing = 0, nothingHosts = 0, partHosts = 0;
  for (const v of vws) {
    const full = Math.max(0, num(v.weight)) / total * minted;
    const got = (rows.find((r) => r.address === v.member_address) || { rewarded: 0 }).rewarded;
    if (full > 0 && got === 0) { nothing += full; nothingHosts++; }
    else if (got < full * 0.98) partHosts++;
  }
  const notPaid = Math.max(0, minted - paid);
  nothing = Math.min(nothing, notPaid);
  return { minted, paid, notPaid, nothing, nothingHosts, rest: notPaid - nothing, partHosts };
}

/**
 * The network page: where the epoch is, who is in it, what they run, what was
 * paid out, what exists, and how all of it moved over the recent epochs.
 */
async function networkOverview(seeds, { epochs = 180 } = {}) {
  const [now, mem, file, kept, cap, use] = await Promise.all([epochNow(seeds), members(seeds), networkFile(), epochArchive(), capacityFile(), usageFile()]);
  const byEpoch = kept || {};
  const totalWeight = mem.hosts.reduce((a, h) => a + h.weight, 0);
  const finished = Object.keys(byEpoch).map(Number).filter((e) => e < mem.epoch).sort((a, b) => a - b);
  const lastEpoch = finished.length ? finished[finished.length - 1] : null;
  const lastRows = lastEpoch ? byEpoch[lastEpoch] : [];
  const sum = (a) => a.reduce((x, y) => x + y, 0);

  // The past, one entry an epoch: how many hosts took part and what was paid.
  const past = finished.slice(-epochs).map((e) => ({ epoch: e, ...epochWhen(e, now), hosts: byEpoch[e].length, rewards: sum(byEpoch[e].map((r) => r.rewarded)) }));

  // Everyone in the current epoch, with what they run and how they did last epoch.
  const files = (file && file.hosts) || {};
  // An epoch with no request counts for anyone: "missed" is unknown, not 0%.
  const lastRecorded = lastRows.some((r) => r.requests + r.missed > 0);
  // AI work through devshards in the newest epoch the data job has results for:
  // each host's missed and wrong answers in its AI sessions, and the middle host's rate.
  const wu = ((use && use.epochs) || []).filter((u) => u.h).sort((a, b) => b.e - a.e)[0] || null;
  const wRows = wu ? Object.values(wu.h).filter((a) => a[5] > 0) : [];
  const typicalMissedPer = wRows.length ? median(wRows.map((a) => a[0] / a[5])) : null;
  const hosts = mem.hosts.map((h) => {
    const g = files[h.address] || null;
    const last = lastRows.find((r) => r.address === h.address) || null;
    const ds = wu && wu.h[h.address];
    return {
      work: ds ? { epoch: wu.e, missed: ds[0], invalid: ds[1], sessions: ds[5], missedPer: ds[5] ? ds[0] / ds[5] : null, typicalMissedPer,
        missedVs: ds[5] ? beside(ds[0] / ds[5], typicalMissedPer, WORK_SLACK.missed, ds[5]) : null } : null,
      address: h.address,
      url: h.url,
      weight: h.weight,
      share: share(h.weight, totalWeight),
      models: h.models,
      gpus: g ? g.gpus : null,
      gpusWorking: g ? g.working : null,
      kinds: g ? g.kinds : null,
      // It reports far fewer GPUs than its weight takes, and its weight in H100s (scripts/network-snapshot.js).
      underReported: !!(g && g.underReported),
      h100: g && g.h100 != null ? g.h100 : null,
      lastReward: last ? last.rewarded : null,
      missedShare: last && lastRecorded ? missedShare(last) : null,
      invalidShare: last && last.validated + last.invalidated > 0 ? invalidShare(last) : null
    };
  }).sort((a, b) => b.weight - a.weight);
  const out = new Set(mem.excluded.map((x) => x.address));

  // Which models the hosts serve, by how many hosts.
  const models = {};
  for (const h of mem.hosts) for (const m of h.models) models[m] = (models[m] || 0) + 1;

  const totals = (file && file.totals) || null;
  const lastRewards = lastRows.length ? sum(lastRows.map((r) => r.rewarded)) : null;
  return {
    epoch: {
      index: now.index, phase: now.phase,
      secondsToNextPoc: Math.max(0, (now.nextPocStart - now.height) * now.secPerBlock),
      progress: now.nextPocStart > now.pocStart ? Math.min(1, Math.max(0, (now.height - now.pocStart) / (now.nextPocStart - now.pocStart))) : 0,
      hours: now.epochBlocks * now.secPerBlock / 3600
    },
    hosts: hosts.map((h) => ({ ...h, excluded: out.has(h.address) })),
    excluded: mem.excluded,
    totalWeight,
    // The GPUs hosts report — not checked by the network — and what is real: the
    // network's weight in H100s, and the hosts whose weight their GPUs cannot make.
    gpus: totals ? { total: totals.gpus, working: totals.gpusWorking, machines: totals.machines, kinds: totals.kinds, countedAt: file.asOf, epoch: file.epoch,
      h100: totals.h100 || null, underReported: totals.underReported || null } : null,
    models: Object.entries(models).map(([model, n]) => ({ model, hosts: n })).sort((a, b) => b.hosts - a.hosts),
    last: lastEpoch ? { epoch: lastEpoch, rewards: lastRewards, hosts: lastRows.length, payout: await soft(epochPayout(seeds, lastEpoch, lastRows)) } : null,
    supply: file ? file.supply : null,
    communityPool: file ? file.communityPool : null,
    past,
    // The GPUs that took part, epoch by epoch (oldest first), and of what kind:
    // measured by GNOT's data job while the epoch was on, or else Gonkascan's record.
    capacity: ((cap && cap.epochs) || []).filter((c) => c.e > now.index - epochs && (c.g != null || c.gs != null)).map((c) => ({
      epoch: c.e, ...epochWhen(c.e, now), gpus: c.g != null ? c.g : c.gs, kinds: c.k || {}, measured: c.g != null
    })),
    // The network's total weight, epoch by epoch (oldest first). What users paid
    // for AI is in the file too but not passed on: an answer still running when
    // an epoch's books close is charged its whole reservation, so it says little.
    usage: ((use && use.epochs) || []).filter((u) => u.e > now.index - epochs).map((u) => ({
      epoch: u.e, ...epochWhen(u.e, now),
      weight: u.w > 0 ? u.w : null
    }))
  };
}

/** Who is on the network right now — to pick a node from, and to compare with. */
async function network(seeds) {
  const mem = await members(seeds);
  const total = mem.hosts.reduce((a, h) => a + h.weight, 0);
  return {
    epoch: mem.epoch,
    hosts: mem.hosts.map((h) => ({ address: h.address, weight: h.weight, share: share(h.weight, total), models: h.models, nodes: h.nodes.length }))
      .sort((a, b) => b.weight - a.weight),
    excluded: mem.excluded
  };
}

/** What a proposal would do, from the kinds of message it carries. */
const WHAT_IT_DOES = {
  MsgSoftwareUpgrade: "upgrade",
  MsgUpdateParams: "settings",
  MsgCommunityPoolSpend: "spend",
  MsgExecuteContract: "contract",
  MsgInstantiateContract2: "contract",
  MsgMigrateContract: "contract",
  MsgBatchTransferWithVesting: "payout",
  MsgTransferWithVesting: "payout",
  MsgRegisterModel: "model-add",
  MsgDeleteGovernanceModel: "model-remove"
};
/** Every ngonka amount anywhere inside a proposal's messages, added up. */
function askedIn(messages) {
  let total = 0;
  (function walk(o) {
    if (Array.isArray(o)) o.forEach(walk);
    else if (o && typeof o === "object") {
      if (/ngonka/i.test(String(o.denom || "")) && o.amount) total += num(o.amount);
      Object.values(o).forEach(walk);
    }
  })(messages);
  return total;
}
/**
 * What a proposal pays, and from where, read from its messages the same way
 * scripts/governance-snapshot.js does: from the community pool, into the
 * community sale, from the governance account (in coins that unlock over
 * time), or in USDT from the community sale. `ctx` names the governance
 * account, the sale contracts and the USDT token (from governance.json).
 */
function payoutsIn(messages, ctx = {}) {
  const coinsOf = (list) => (list || []).reduce((a, c) => (/^ngonka$/i.test(c.denom) ? a + num(c.amount) : a), 0);
  const usdtOf = (list) => (ctx.usdtDenom ? (list || []).reduce((a, c) => (c.denom === ctx.usdtDenom ? a + num(c.amount) : a), 0) : 0);
  const sale = new Set(ctx.sale || []);
  const out = { pool: 0, toSale: 0, gov: 0, usdt: 0, poolUsdt: 0, recipients: 0 };
  const to = new Set();
  for (const m of messages || []) {
    const type = String(m["@type"] || "").split(".").pop();
    if (type === "MsgCommunityPoolSpend") {
      // GNK sent from the pool to the governance account only moves it there;
      // the governance account's own payout of it is counted below, once.
      if (ctx.gov && m.recipient === ctx.gov) continue;
      const a = coinsOf(m.amount), u = usdtOf(m.amount);
      if (sale.has(m.recipient)) out.toSale += a;
      else { out.pool += a; out.poolUsdt += u; if (a || u) to.add(m.recipient); }
    } else if (type === "MsgBatchTransferWithVesting" && (!ctx.gov || m.sender === ctx.gov)) {
      for (const o of m.outputs || []) { out.gov += coinsOf(o.amount); to.add(o.recipient); }
    } else if (type === "MsgTransferWithVesting" && (!ctx.gov || m.sender === ctx.gov || m.from_address === ctx.gov)) {
      out.gov += coinsOf(m.amount); to.add(m.recipient || m.to_address);
    } else if (type === "MsgMultiSend" && (!ctx.gov || (m.inputs || []).some((i) => i.address === ctx.gov))) {
      for (const o of m.outputs || []) { out.gov += coinsOf(o.coins); to.add(o.address); }
    } else if (type === "MsgExecuteContract" && m.msg && m.msg.withdraw_ibc && (!ctx.usdtDenom || m.msg.withdraw_ibc.denom === ctx.usdtDenom)) {
      out.usdt += num(m.msg.withdraw_ibc.amount); to.add(m.msg.withdraw_ibc.recipient);
    }
  }
  to.delete(undefined);
  out.recipients = to.size;
  return out;
}

function tidyProposal(p, { totalWeight, quorum, threshold, liveTally, voters, ctx }) {
  const t = liveTally || p.final_tally_result || {};
  const yes = num(t.yes_count), no = num(t.no_count), veto = num(t.no_with_veto_count), abstain = num(t.abstain_count);
  const voted = yes + no + veto + abstain;
  const kinds = [...new Set((p.messages || []).map((m) => String(m["@type"] || "").split(".").pop()))];
  const upgrade = (p.messages || []).find((m) => /MsgSoftwareUpgrade$/.test(String(m["@type"])));
  return {
    id: num(p.id),
    title: p.title || "",
    summary: p.summary || "",
    status: String(p.status || "").replace("PROPOSAL_STATUS_", ""),      // VOTING_PERIOD, PASSED, REJECTED, FAILED, DEPOSIT_PERIOD
    submitted: p.submit_time || null, starts: p.voting_start_time || null, ends: p.voting_end_time || null,
    proposer: p.proposer || "",
    does: [...new Set(kinds.map((k) => WHAT_IT_DOES[k] || "other"))],
    kinds,
    asked: askedIn(p.messages),
    pays: payoutsIn(p.messages, ctx),
    upgradeTo: upgrade && upgrade.plan ? { name: upgrade.plan.name, height: num(upgrade.plan.height) } : null,
    tally: { yes, no, veto, abstain, voted },
    // How many wallets voted, weight or not (live votes only; null when unknown).
    voters: Number.isFinite(voters) ? voters : null,
    // Of those who voted yes or no; abstaining counts toward turnout only.
    yesShare: share(yes, yes + no + veto),
    turnout: totalWeight > 0 ? share(voted, totalWeight) : null,
    quorum, threshold
  };
}

/** The newest proposals, live ones first, with how the vote stands. */
/**
 * Every proposal there has ever been, newest first. All of them (about 800 KB
 * for the first 115) are read at most every half hour; in between only the
 * newest few are read again, which is where votes and outcomes still change.
 */
let propCache = { at: 0, list: null };
async function everyProposal(seeds) {
  const page = (key, limit) => ask(seeds, `/chain-api/cosmos/gov/v1/proposals?pagination.reverse=true&pagination.limit=${limit}` +
    (key ? "&pagination.key=" + encodeURIComponent(key) : ""), 60000);
  if (!propCache.list || Date.now() - propCache.at > 30 * 60 * 1000) {
    const all = [];
    let key = null;
    for (let i = 0; i < 20; i++) {
      const r = await page(key, 200);
      all.push(...((r && r.proposals) || []));
      key = r && r.pagination && r.pagination.next_key;
      if (!key) break;
    }
    propCache = { at: Date.now(), list: all };
    return all;
  }
  const fresh = ((await page(null, 12)) || {}).proposals || [];
  const byId = new Map(propCache.list.map((p) => [String(p.id), p]));
  for (const p of fresh) byId.set(String(p.id), p);
  propCache.list = [...byId.values()].sort((a, b) => num(b.id) - num(a.id));
  return propCache.list;
}

/**
 * All the voting power there is: what the chain measures a vote's turnout
 * against, the bonded validators' power. Hosts' epoch weights add up to less
 * (Nov 9, 2026: 2,048,550 against 2,233,602), so a turnout measured against
 * them read too high: 10% where the chain counted 9.4%. Gonka's staking pool
 * reports 0, so the validator set is added up instead. 0 when unknown.
 */
async function votingPowerTotal(seeds) {
  let total = 0, seen = 0;
  for (let i = 0; i < 10; i++) {
    const r = await ask(seeds, `/chain-api/cosmos/base/tendermint/v1beta1/validatorsets/latest?pagination.limit=200&pagination.offset=${seen}`);
    const vals = (r && r.validators) || [];
    total += vals.reduce((a, v) => a + num(v.voting_power), 0);
    seen += vals.length;
    if (!vals.length || seen >= num(r && r.pagination && r.pagination.total)) break;
  }
  return total;
}

/** How many wallets have voted on a proposal, whatever their weight. */
async function votersOf(seeds, id) {
  let n = 0, key = "";
  for (let i = 0; i < 10; i++) {
    const r = await ask(seeds, `/chain-api/cosmos/gov/v1/proposals/${id}/votes?pagination.limit=200` + (key ? "&pagination.key=" + encodeURIComponent(key) : ""));
    n += ((r && r.votes) || []).length;
    key = r && r.pagination && r.pagination.next_key;
    if (!key) break;
  }
  return n;
}

async function proposals(seeds) {
  const [list, params, mem, money, power] = await Promise.all([
    everyProposal(seeds),
    soft(ask(seeds, "/chain-api/cosmos/gov/v1/params/tallying")),
    soft(members(seeds)),
    governanceFile(),
    soft(votingPowerTotal(seeds))
  ]);
  const ctx = money ? { gov: money.govAccount && money.govAccount.address, sale: (money.sale || []).map((s) => s.address), usdtDenom: money.usdt && money.usdt.denom } : {};
  const hostsWeight = mem ? mem.hosts.reduce((a, h) => a + h.weight, 0) : 0;
  // Turnout is measured as the chain measures it; the hosts' weights only stand in when that cannot be read.
  const totalWeight = power > 0 ? power : hostsWeight;
  const tp = (params && params.params) || {};
  const rules = { quorum: num(tp.quorum), threshold: num(tp.threshold), veto: num(tp.veto_threshold) };
  const raw = list || [];
  const out = [];
  for (const p of raw) {
    const live = /VOTING_PERIOD/.test(String(p.status));
    const [tally, voters] = live ? await Promise.all([soft(ask(seeds, `/chain-api/cosmos/gov/v1/proposals/${p.id}/tally`)), soft(votersOf(seeds, p.id))]) : [null, undefined];
    out.push(tidyProposal(p, { totalWeight, quorum: rules.quorum, threshold: rules.threshold, liveTally: tally && tally.tally, voters, ctx }));
  }
  out.sort((a, b) => (Number(b.status === "VOTING_PERIOD") - Number(a.status === "VOTING_PERIOD")) || (b.id - a.id));
  return { rules, totalWeight, weightFrom: power > 0 ? "chain" : "hosts", hosts: mem ? mem.hosts.length : 0, proposals: out, money: money || null };
}

/** How one address voted on a proposal that is still open (the chain forgets votes afterwards). */
async function voteOf(seeds, id, address) {
  const j = await soft(ask(seeds, `/chain-api/cosmos/gov/v1/proposals/${id}/votes/${address}`));
  const opt = j && j.vote && ((j.vote.options || [])[0] || {}).option;
  return opt ? String(opt).replace("VOTE_OPTION_", "") : null;        // YES, NO, ABSTAIN, NO_WITH_VETO
}

const looksLikeAddress = (s) => /^gonka1[02-9ac-hj-np-z]{38}$/.test(String(s || "").trim());

/* The nodes a person has chosen to watch: an address and the name they gave
   it, kept in a small file on this computer and nowhere else. A node of
   their own also keeps the name of its wallet in this computer's keyring and
   how to reach its server. Never a password or a passphrase: those are asked
   for each time. */
const MAX_NODES = 50;
const tidyText = (s, max) => String(s || "").replace(/[\x00-\x1f\x7f]/g, "").trim().slice(0, max);
function tidyServer(s) {
  if (!s || typeof s !== "object") return null;
  if (s.mode === "local") return { mode: "local" };
  const host = tidyText(s.host, 253);
  const username = tidyText(s.username, 64);
  if (!host || !username || /\s/.test(host)) return null;
  const port = Math.round(Number(s.port) || 22);
  const auth = s.auth === "password" ? "password" : "key";
  const out = { mode: "ssh", host, port: port >= 1 && port <= 65535 ? port : 22, username, auth };
  if (auth === "key" && s.privateKeyPath) out.privateKeyPath = tidyText(s.privateKeyPath, 1024);
  return out;
}
function tidyNode(n) {
  const out = { address: String(n.address).trim(), name: tidyText(n.name, 40) };
  const wallet = tidyText(n.wallet, 64);
  if (wallet) out.wallet = wallet;
  const server = tidyServer(n.server);
  if (server) out.server = server;
  return out;
}
function loadNodes(dir) {
  try {
    const list = JSON.parse(fs.readFileSync(path.join(dir, "nodes.json"), "utf8"));
    return Array.isArray(list) ? list.filter((n) => n && looksLikeAddress(n.address)).slice(0, MAX_NODES).map(tidyNode) : [];
  } catch (_) { return []; }
}
function saveNodes(dir, nodes) {
  const seen = new Set();
  const clean = [];
  for (const n of Array.isArray(nodes) ? nodes : []) {
    const address = String((n && n.address) || "").trim();
    if (!looksLikeAddress(address) || seen.has(address)) continue;
    seen.add(address);
    clean.push(tidyNode({ ...n, address }));
    if (clean.length >= MAX_NODES) break;
  }
  fs.mkdirSync(dir, { recursive: true });
  fs.writeFileSync(path.join(dir, "nodes.json"), JSON.stringify(clean, null, 1));
  return clean;
}

/**
 * The votes cast from this computer, newest first: votes.json beside
 * nodes.json. The chain only keeps a vote while its proposal is open, so
 * without this a host could not see afterwards how they voted. One entry per
 * proposal and address (a changed vote replaces the earlier one); no secret
 * is in it, only what the chain itself showed while the vote was open.
 */
const VOTE_OPTIONS = ["YES", "NO", "NO_WITH_VETO", "ABSTAIN"];
const MAX_VOTES = 2000;
function loadVotes(dir) {
  try {
    const list = JSON.parse(fs.readFileSync(path.join(dir, "votes.json"), "utf8"));
    return Array.isArray(list)
      ? list.filter((v) => v && Number(v.id) > 0 && looksLikeAddress(v.address) && VOTE_OPTIONS.includes(v.option)).slice(0, MAX_VOTES)
      : [];
  } catch (_) { return []; }
}
function rememberVote(dir, v) {
  const id = Number(v && v.id), address = String((v && v.address) || "").trim(), option = String((v && v.option) || "");
  if (!(id > 0) || !looksLikeAddress(address) || !VOTE_OPTIONS.includes(option)) throw new Error("That is not a vote.");
  const txhash = /^[0-9A-Fa-f]{64}$/.test(String((v && v.txhash) || "")) ? String(v.txhash).toUpperCase() : undefined;
  const entry = { id, address, option, at: Date.now(), ...(txhash ? { txhash } : {}) };
  const rest = loadVotes(dir).filter((x) => !(Number(x.id) === id && x.address === address));
  const list = [entry, ...rest].slice(0, MAX_VOTES);
  fs.mkdirSync(dir, { recursive: true });
  fs.writeFileSync(path.join(dir, "votes.json"), JSON.stringify(list, null, 1));
  return list;
}

const SEEDS = KNOWLEDGE.seedNodes;
const KEEP_MS = 60 * 1000;
const kept = new Map();          // question -> { at, status, data } or { pending }

// The price as the app has it: the middle of the markets, read as the app
// reads them, from the visitor's own computer. The page reads the markets in
// the visitor's browser (read-markets.js) and sends them as "markets"; the
// markets limit how often one address may ask, and this server shares its
// address with many other websites, so they often refuse it. A market the
// browser could not read is taken from this server's own reading, and else
// from the data job's last reading of it in earnings.json (scripts/base44-price.js).
const EARNINGS_URL = KNOWLEDGE.epochsUrl.replace(/epochs\.json$/, "earnings.json");
let published = { at: 0, p: null };
async function publishedPrice() {
  if (!published.p || Date.now() - published.at > 10 * 60 * 1000) {
    try {
      const f = await (await fetch(EARNINGS_URL, { signal: AbortSignal.timeout(15000) })).json();
      if (f.price && f.price.usd) published = { at: Date.now(), p: { ...f.price, asOf: f.asOf } };
    } catch (_) { /* keep the last one */ }
  }
  return published.p;
}
// The markets the app reads, in its order (src/services/earnings.js gnkPrice).
const MARKETS = ["CoinGecko","CoinPaprika"];
function fillInPrice(live, published) {
  const have = new Map(((live && live.sources) || []).map((s) => [s.name, s]));
  const answered = have.size;
  for (const s of (published && published.sources) || []) {
    if (!have.has(s.name) && s.usd > 0) have.set(s.name, { name: s.name, usd: s.usd, readAt: published.asOf });
  }
  const sources = [...have.values()];
  if (!sources.length) return { usd: null, sources: [] };
  const sorted = sources.map((s) => s.usd).sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  const usd = sorted.length % 2 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
  return answered ? { usd, sources } : { usd, sources, published: published && published.asOf };
}
function marketsFrom(body) {
  const seen = new Map();
  for (const m of (body && Array.isArray(body.markets) ? body.markets : [])) {
    const usd = Number(m && m.usd);
    if (m && MARKETS.includes(m.name) && usd > 0 && isFinite(usd) && !seen.has(m.name)) seen.set(m.name, { name: m.name, usd });
  }
  return [...seen.values()];
}
function priceFrom(browser, own, published) {
  const have = new Map(browser.map((s) => [s.name, s]));
  for (const s of (own && own.sources) || []) if (!have.has(s.name)) have.set(s.name, { name: s.name, usd: s.usd });
  const p = fillInPrice({ sources: [...have.values()] }, published);
  p.sources.sort((a, b) => MARKETS.indexOf(a.name) - MARKETS.indexOf(b.name));
  return p;
}
/** This server's own reading of the markets, kept five minutes (one when none answered). */
async function ownPrice() {
  if (own.p && Date.now() - own.at < (own.p.sources.length ? 5 * 60 * 1000 : 60 * 1000)) return own.p;
  let live = { usd: null, sources: [] };
  try { live = await gnkPrice(); } catch (_) { /* none answered */ }
  own = { at: Date.now(), p: live };
  return live;
}
/** The price for one visitor: what their browser read, then this server's reading, then the data job's. */
async function priceNow(browser) {
  const mine = browser.length < MARKETS.length ? await ownPrice() : null;
  return priceFrom(browser, mine, await publishedPrice());
}

/** One answer per question a minute; visitors asking at the same moment share it. */
async function once(key, work) {
  const k = kept.get(key);
  if (k && k.pending) return k.pending;
  if (k && Date.now() - k.at < KEEP_MS) return k;
  const pending = (async () => {
    try { return { at: Date.now(), status: 200, data: await work() }; }
    catch (e) { return { at: Date.now(), status: 502, data: { error: String((e && e.message) || e) } }; }
  })();
  kept.set(key, { pending });
  const done = await pending;
  // A failure is not kept: the next visitor asks again.
  if (done.status === 200) kept.set(key, done); else kept.delete(key);
  if (kept.size > 300) kept.delete(kept.keys().next().value);
  return done;
}

Deno.serve(async (req) => {
  const headers = { "Content-Type": "application/json", "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "*", "Cache-Control": "public, max-age=60" };
  if (req.method === "OPTIONS") return new Response(null, { headers });
  const q = Object.fromEntries(new URL(req.url).searchParams);
  if (req.method === "POST") { try { Object.assign(q, await req.json()); } catch (_) { /* no body */ } }
  const view = String(q.view || "network");
  const address = String(q.address || "").trim();
  const wrong = (why) => new Response(JSON.stringify({ error: why }), { status: 400, headers });
  // The network's data is shared by every visitor for a minute; the price is each visitor's own.
  const priced = async (key, work) => {
    const [a, price] = await Promise.all([once(key, work), priceNow(marketsFrom(q))]);
    return a.status === 200 ? { status: 200, data: { ...a.data, price } } : a;
  };
  let a;
  if (view === "price") {
    a = { status: 200, data: { price: await priceNow(marketsFrom(q)) } };
  } else if (view === "network") {
    a = await priced("network", () => Monitor.networkOverview(SEEDS));
  } else if (view === "votes") {
    a = await once("votes", () => Monitor.proposals(SEEDS));
  } else if (view === "node") {
    if (!Monitor.looksLikeAddress(address)) return wrong("That is not a Gonka address.");
    const epochs = Math.min(180, Math.max(7, Number(q.epochs) || 30));
    a = await priced(`node|${address}|${epochs}`, () => Monitor.overview(SEEDS, address, { epochs }));
  } else if (view === "answers") {
    // The addresses are the ones the hosts registered, read here: a visitor cannot name one.
    a = await once("answers", async () => {
      const mem = await Monitor.members(SEEDS);
      const got = await Promise.all(mem.hosts.map((h) => Monitor.knock(h.url, h.address)));
      return {
        epoch: mem.epoch,
        answers: Object.fromEntries(mem.hosts.map((h, i) => [h.address, got[i] ? got[i].ok : null])),
        // The HTTP status each address answered last, null for none: what this server's gateway hands back.
        status: Object.fromEntries(mem.hosts.map((h, i) => [h.address, got[i] ? got[i].status : null])),
        // When each answer was found out: null = just now, by this server; a time = the data job's check from GitHub.
        checkedAt: Object.fromEntries(mem.hosts.map((h, i) => [h.address, got[i] && got[i].seenAt ? got[i].seenAt : null]))
      };
    });
  } else if (view === "vote") {
    if (!Monitor.looksLikeAddress(address)) return wrong("That is not a Gonka address.");
    const id = Math.round(Number(q.id));
    if (!(id > 0)) return wrong("That is not a proposal number.");
    a = await once(`vote|${id}|${address}`, async () => ({ vote: await Monitor.voteOf(SEEDS, id, address) }));
  } else if (view === "errors") {
    // What went wrong lately when this server asked Gonka's public nodes. Public questions only; nothing is kept for long.
    a = { status: 200, data: { at: new Date().toISOString(), ...Monitor.recentTrouble() } };
  } else return wrong("Ask for view=price, network, votes, node, vote, answers or errors.");
  return new Response(JSON.stringify(a.data), { status: a.status, headers });
});
